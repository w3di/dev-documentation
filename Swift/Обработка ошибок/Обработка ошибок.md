## Error protocol

Любой тип, подписанный на `Error`, может быть брошен:

```swift
enum NetworkError: Error {
    case noConnection
    case timeout(seconds: Int)
    case httpError(statusCode: Int, message: String)
    case decodingFailed(underlying: Error)
}

// Struct тоже может быть ошибкой
struct ValidationError: Error {
    let field: String
    let message: String
}
```

### LocalizedError

```swift
enum AppError: LocalizedError {
    case fileNotFound(path: String)
    case permissionDenied

    var errorDescription: String? {
        switch self {
        case .fileNotFound(let path):
            return "File not found: \(path)"
        case .permissionDenied:
            return "Permission denied"
        }
    }

    var recoverySuggestion: String? {
        switch self {
        case .fileNotFound:
            return "Check the file path and try again"
        case .permissionDenied:
            return "Run with elevated privileges"
        }
    }
}
```

## throws и do-catch

```swift
func parseJSON(_ data: Data) throws -> Config {
    guard !data.isEmpty else {
        throw NetworkError.noConnection
    }
    return try JSONDecoder().decode(Config.self, from: data)
}

// do-catch
do {
    let config = try parseJSON(data)
    print(config)
} catch NetworkError.noConnection {
    print("No connection")
} catch NetworkError.httpError(let code, let msg) where code >= 500 {
    print("Server error \(code): \(msg)")
} catch let error as NetworkError {
    print("Network error: \(error)")
} catch {
    // error — автоматическая переменная типа any Error
    print("Unknown error: \(error)")
}
```

### throws в инициализаторах

```swift
struct Config {
    let apiKey: String
    let endpoint: URL

    init(json: [String: Any]) throws {
        guard let key = json["apiKey"] as? String else {
            throw ValidationError(field: "apiKey", message: "Missing")
        }
        guard let urlString = json["endpoint"] as? String,
              let url = URL(string: urlString) else {
            throw ValidationError(field: "endpoint", message: "Invalid URL")
        }
        self.apiKey = key
        self.endpoint = url
    }
}
```

## try варианты

### try — пробрасывает ошибку

```swift
func process() throws {
    let data = try fetchData()    // ошибка пробрасывается выше
    let result = try parse(data)  // ошибка пробрасывается выше
}
```

### try? — конвертирует в Optional

```swift
let config = try? parseJSON(data)  // Config? (nil если ошибка)

// Полезно с guard
guard let config = try? loadConfig() else {
    useDefaultConfig()
    return
}

// Полезно с ??
let config = (try? loadConfig()) ?? Config.default
```

### try! — force (crash при ошибке)

```swift
// Используй только когда 100% уверен, что ошибки не будет
let bundleConfig = try! loadConfig(from: Bundle.main)

// Аналогично force unwrap — crash в production при ошибке
```

## Typed throws (Swift 6+)

Указание конкретного типа ошибки:

```swift
func parse(data: Data) throws(ParseError) -> Document {
    guard let string = String(data: data, encoding: .utf8) else {
        throw .invalidEncoding
    }
    // ...
}

// Компилятор знает точный тип ошибки
do {
    let doc = try parse(data: data)
} catch .invalidEncoding {
    // Можно матчить без приведения типа
}

// throws(Never) — гарантия, что не бросает
func safeDivide(_ a: Int, _ b: Int) throws(Never) -> Double {
    // Не может бросить ошибку
    Double(a) / Double(b)
}
```

## Rethrows

Функция бросает ошибку **только если** переданное замыкание бросает:

```swift
func retry<T>(times: Int, task: () throws -> T) rethrows -> T {
    for attempt in 1..<times {
        do {
            return try task()
        } catch {
            print("Attempt \(attempt) failed")
        }
    }
    return try task()  // последняя попытка — пробрасываем ошибку
}

// Вызов с throwing closure — нужен try
let result = try retry(times: 3) { try riskyOperation() }

// Вызов с non-throwing closure — не нужен try
let safe = retry(times: 3) { 42 }
```

## Result

```swift
enum Result<Success, Failure: Error> {
    case success(Success)
    case failure(Failure)
}

// Создание
func fetchUser(id: Int) -> Result<User, NetworkError> {
    guard id > 0 else {
        return .failure(.httpError(statusCode: 400, message: "Invalid ID"))
    }
    return .success(User(id: id, name: "Alice"))
}

// Использование
let result = fetchUser(id: 42)

switch result {
case .success(let user):
    print(user.name)
case .failure(let error):
    print(error)
}

// Преобразования
let mapped = result.map { $0.name }           // Result<String, NetworkError>
let flatMapped = result.flatMap { validate($0) }  // Result<ValidUser, NetworkError>

// Result ↔ throws
let user = try result.get()  // бросает Failure если .failure

let result2 = Result { try parseJSON(data) }  // Result<Config, Error>
```

## Assertions

### assert — только в debug

```swift
func divide(_ a: Int, _ b: Int) -> Int {
    assert(b != 0, "Division by zero")  // убирается в release
    return a / b
}
```

### precondition — debug + release (не в -Ounchecked)

```swift
func element(at index: Int) -> Element {
    precondition(index >= 0 && index < count, "Index out of bounds")
    return storage[index]
}
```

### fatalError — всегда crash

```swift
func requiredMethod() {
    fatalError("Subclass must implement this method")
}

// В switch для невозможных случаев
switch impossible {
default:
    fatalError("This should never happen")
}
```

| | Debug | Release | -Ounchecked |
|--|-------|---------|-------------|
| `assert` | Crash | Пропускается | Пропускается |
| `precondition` | Crash | Crash | Пропускается |
| `fatalError` | Crash | Crash | Crash |

## Never

Тип, обозначающий «функция никогда не возвращает»:

```swift
func crashAndBurn() -> Never {
    fatalError("💥")
}

// Полезно в switch
func handle(_ code: Int) -> String {
    switch code {
    case 200: return "OK"
    case 404: return "Not Found"
    default: fatalError("Unhandled code: \(code)")  // Never — компилятор знает, что return не нужен
    }
}

// Never как bottom type
// Result<Int, Never> — результат, который не может быть failure
let alwaysSuccess: Result<Int, Never> = .success(42)
let value = try! alwaysSuccess.get()  // безопасно — failure невозможен
```

## Производительность

Swift error handling использует **return-based** реализацию (не exception unwinding):

```swift
// Внутренне throws → возврат Result-подобной структуры
func parse() throws -> Data { /* ... */ }

// Компилятор генерирует примерно:
// func parse() -> (Data?, Error?) { ... }
```

- **Happy path (нет ошибки):** практически **zero cost** — обычный return
- **Error path:** аллокация Error на heap + return
- **Нет stack unwinding** как в C++ exceptions
- **Нет таблиц исключений** как в Java

## Сравнение с JS

| Аспект | Swift | JavaScript |
|--------|-------|-----------|
| Error type | `Error` protocol (любой тип) | `Error` class (или что угодно) |
| Throwing | `throws` в сигнатуре | Любая функция может бросить |
| Catching | `do-catch` (исчерпывающий) | `try-catch` (один catch) |
| Optional try | `try?` → Optional | Нет аналога |
| Typed errors | `throws(MyError)` (Swift 6) | Нет |
| Result type | `Result<T, E>` | Нет (Promise — близко) |
| Compile-time | Компилятор заставляет обрабатывать | Нет проверок |
| Performance | Zero-cost happy path | Stack unwinding |

```swift
// Swift — компилятор заставляет обрабатывать
let data = try fetchData()  // ошибка компиляции без try/do-catch

// JS — никаких проверок
// const data = fetchData()  // может бросить, компилятор молчит
// const data = await fetchData()  // Unhandled Promise rejection
```
