## Swift Regex (5.7+)

Swift 5.7 представил first-class поддержку regex с **типизированными captures**:

```swift
import RegexBuilder

// Два способа создания regex:
// 1. Regex литерал
let pattern = /\d{3}-\d{3}-\d{4}/

// 2. RegexBuilder DSL
let phone = Regex {
    Repeat(.digit, count: 3)
    "-"
    Repeat(.digit, count: 3)
    "-"
    Repeat(.digit, count: 4)
}

// Использование
let text = "Call me at 555-123-4567"
if let match = text.firstMatch(of: pattern) {
    print(match.output)  // "555-123-4567"
}
```

## Regex литералы

```swift
// Простой паттерн
let digits = /\d+/

// С флагами
let caseInsensitive = /hello/i  // case insensitive (нет, используй .ignoresCase())

// Правильно — флаги через модификатор
let pattern = /hello/.ignoresCase()

// Escaping
let url = /https?:\/\/[\w\.-]+/

// Extended literal (многострочный)
let complex = #/
    (\d{4})   # год
    -
    (\d{2})   # месяц
    -
    (\d{2})   # день
/#

// Regex<(Substring, Substring, Substring, Substring)>
// Tuple: (whole match, year, month, day)
```

### Regex<Output> — типизированный

```swift
let datePattern = /(\d{4})-(\d{2})-(\d{2})/
// Тип: Regex<(Substring, Substring, Substring, Substring)>

if let match = "2024-01-15".firstMatch(of: datePattern) {
    let year = match.1   // Substring "2024"
    let month = match.2  // Substring "01"
    let day = match.3    // Substring "15"
}
```

## RegexBuilder

DSL для создания сложных regex с читаемым синтаксисом:

```swift
import RegexBuilder

// Базовые компоненты
let pattern = Regex {
    "Hello"                    // литерал
    One(.whitespace)           // один пробел
    OneOrMore(.word)           // один или более word chars
}

// Quantifiers
let flexible = Regex {
    ZeroOrMore(.digit)         // *
    OneOrMore(.digit)          // +
    Optionally("suffix")       // ?
    Repeat(.digit, count: 3)   // {3}
    Repeat(.digit, 2...5)      // {2,5}
}

// Alternation
let greeting = Regex {
    ChoiceOf {
        "Hello"
        "Hi"
        "Hey"
    }
    ", "
    OneOrMore(.word)
}

// Character classes
let custom = Regex {
    One(.anyOf("aeiou"))           // [aeiou]
    One(.anyNonNewline)            // .
    CharacterClass(.digit, .anyOf("-"))  // [\d-]
}
```

### Captures с типами

```swift
let logPattern = Regex {
    Capture {
        ChoiceOf {
            "INFO"
            "WARN"
            "ERROR"
        }
    }
    ": "
    Capture {
        OneOrMore(.anyNonNewline)
    }
}
// Тип: Regex<(Substring, Substring, Substring)>

if let match = "ERROR: Something failed".firstMatch(of: logPattern) {
    let level = match.1   // "ERROR"
    let message = match.2 // "Something failed"
}
```

### TryCapture — с преобразованием

```swift
let dateParser = Regex {
    TryCapture {
        Repeat(.digit, count: 4)
    } transform: {
        Int($0)  // Substring → Int?
    }
    "-"
    TryCapture {
        Repeat(.digit, count: 2)
    } transform: {
        Int($0)
    }
    "-"
    TryCapture {
        Repeat(.digit, count: 2)
    } transform: {
        Int($0)
    }
}
// Тип: Regex<(Substring, Int, Int, Int)>

if let match = "2024-01-15".firstMatch(of: dateParser) {
    let year: Int = match.1   // 2024 (Int, не String!)
    let month: Int = match.2  // 1
    let day: Int = match.3    // 15
}
```

### Named captures (Reference)

```swift
let ref = Reference(Substring.self)

let repeated = Regex {
    Capture(as: ref) {
        OneOrMore(.word)
    }
    " "
    ref  // back reference — тот же текст
}

"hello hello".firstMatch(of: repeated)  // match!
"hello world".firstMatch(of: repeated)  // nil
```

## Типизированные captures

Ключевое преимущество Swift Regex — captures имеют **конкретные типы**:

```swift
// Regex литерал — captures как Substring
let pattern1 = /(\d+)\s+(\w+)/
// Regex<(Substring, Substring, Substring)>

// RegexBuilder с transform — captures как Swift типы
let pattern2 = Regex {
    TryCapture { OneOrMore(.digit) } transform: { Int($0) }
    One(.whitespace)
    Capture { OneOrMore(.word) }
}
// Regex<(Substring, Int, Substring)>

if let m = "42 items".firstMatch(of: pattern2) {
    let count: Int = m.1         // 42 (типизировано!)
    let label: Substring = m.2   // "items"
}
```

## Методы

### firstMatch

```swift
let text = "Order #12345 placed on 2024-01-15"
let orderPattern = /#(\d+)/

if let match = text.firstMatch(of: orderPattern) {
    print(match.output.0)  // "#12345" (весь match)
    print(match.output.1)  // "12345" (capture group)
    print(match.range)     // Range<String.Index>
}
```

### wholeMatch

```swift
let emailPattern = /[\w.]+@[\w]+\.[\w]+/

"user@example.com".wholeMatch(of: emailPattern)   // match
"Hello user@example.com".wholeMatch(of: emailPattern)  // nil (не весь текст)
```

### prefixMatch

```swift
"123abc".prefixMatch(of: /\d+/)  // match "123"
"abc123".prefixMatch(of: /\d+/)  // nil (не начинается с цифр)
```

### matches(of:) — все совпадения

```swift
let text = "Scores: 85, 92, 78, 95"
let digits = /\d+/

for match in text.matches(of: digits) {
    print(match.output)  // "85", "92", "78", "95"
}
```

### replacing

```swift
var text = "Hello World"

// Замена первого совпадения
let result = text.replacing(/World/, with: "Swift")  // "Hello Swift"

// Замена всех
let cleaned = "a1b2c3".replacing(/\d/, with: "")  // "abc"

// С transform
let doubled = "price: 100".replacing(/\d+/) { match in
    String(Int(match.output)! * 2)
}  // "price: 200"
```

### split

```swift
let csv = "name,age,city"
let fields = csv.split(separator: /,\s*/)  // ["name", "age", "city"]

let log = "2024-01-15 ERROR Something"
let parts = log.split(separator: /\s+/, maxSplits: 2)
// ["2024-01-15", "ERROR", "Something"]
```

### contains / ranges

```swift
let text = "Hello, World!"

text.contains(/\d+/)         // false
text.ranges(of: /\w+/)       // [Range<String.Index>] для каждого слова

// starts(with:)
"12345".starts(with: /\d+/)  // true
```

## Производительность

### Backtracking

```swift
// ⚠️ Catastrophic backtracking (ReDoS)
// Паттерн (a+)+ на "aaaaaaaaaaX" — экспоненциальное время
let dangerous = /(a+)+/  // Не делай так!

// Atomic groups — предотвращают backtracking
let safe = Regex {
    Local {  // atomic group — нет backtracking внутри
        OneOrMore("a")
    }
}
```

### Possessive quantifiers

```swift
// Possessive (не отдаёт символы обратно при backtracking)
let pattern = Regex {
    OneOrMore(.digit, .possessive)  // не откатывается
    "end"
}
```

### Lazy quantifiers

```swift
// Lazy — минимальное совпадение
let lazy = Regex {
    OneOrMore(.anyNonNewline, .reluctant)  // *? эквивалент
}

// В литерале
let lazyLiteral = /".+?"/  // минимальное совпадение между кавычками
```

## Сравнение с JS

| Аспект | Swift Regex | JavaScript RegExp |
|--------|-----------|------------------|
| Типизация | `Regex<(Substring, Int, ...)>` | Нет (всё string) |
| Создание | Литерал `/pattern/` + Builder DSL | Литерал `/pattern/` + `new RegExp()` |
| Captures | Типизированные (compile-time) | `match[1]` (string, runtime) |
| Named groups | `Reference` / `(?<name>)` | `(?<name>)` (ES2018) |
| Builder | `Regex { }` DSL | Нет |
| Flags | Методы (`.ignoresCase()`) | `/pattern/gi` |
| Global search | `matches(of:)` | `/pattern/g` + `matchAll()` |
| Replace | `.replacing(_:with:)` | `.replace()` / `.replaceAll()` |
| Performance | Atomic groups, possessive | Нет атомарных групп |
| Type safety | Compile-time проверка | Runtime ошибки |

```swift
// Swift — типизированные captures
let pattern = Regex {
    TryCapture { OneOrMore(.digit) } transform: { Int($0) }
    "-"
    Capture { OneOrMore(.word) }
}
if let m = "42-items".firstMatch(of: pattern) {
    let count: Int = m.1        // Int, не String!
    let name: Substring = m.2
}

// JS — всё строки
// const m = "42-items".match(/(\d+)-(\w+)/)
// const count = parseInt(m[1])  // ручная конвертация
// const name = m[2]             // string
```

### Миграция с JS regex

```javascript
// JS
const emailRegex = /^[\w.]+@[\w]+\.[\w]{2,}$/i;
emailRegex.test("user@example.com");
```

```swift
// Swift — литерал
let emailRegex = /^[\w.]+@[\w]+\.[\w]{2,}$/
    .ignoresCase()
"user@example.com".wholeMatch(of: emailRegex) != nil

// Swift — Builder (более читаемый)
let emailBuilder = Regex {
    Anchor.startOfSubject
    OneOrMore(.word, .anyOf("."))
    "@"
    OneOrMore(.word)
    "."
    Repeat(.word, 2...)
    Anchor.endOfSubject
}
.ignoresCase()
```
