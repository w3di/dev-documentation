## switch

В Swift `switch` **исчерпывающий** (exhaustive) — должны быть покрыты все варианты:

```swift
let value = 42

switch value {
case 0:
    print("Zero")
case 1...10:
    print("1 to 10")
case let x where x > 100:
    print("Big: \(x)")
default:
    print("Other")
}

// Нет implicit fallthrough (в отличие от C/JS)
// Каждый case — отдельный scope
// Для fallthrough — explicit keyword
switch value {
case 0:
    print("Zero")
    fallthrough  // переходит к следующему case
case 1:
    print("Zero or One")
default:
    break
}
```

## Типы паттернов

### Wildcard pattern (_)

```swift
let (x, _, z) = (1, 2, 3)  // игнорируем второй элемент
print(x, z)  // 1 3

switch (200, "OK") {
case (200, _):
    print("Success")  // любое сообщение
case (_, "Error"):
    print("Error with any code")
default:
    break
}
```

### Value binding (let/var)

```swift
let point = (3, -2)

switch point {
case (let x, 0):
    print("On x-axis at \(x)")
case (0, let y):
    print("On y-axis at \(y)")
case let (x, y):
    print("At (\(x), \(y))")
}
```

### Tuple patterns

```swift
let httpResponse = (statusCode: 404, message: "Not Found")

switch httpResponse {
case (200..<300, _):
    print("Success")
case (400, let msg):
    print("Bad request: \(msg)")
case (404, _):
    print("Not found")
case (500..., let msg):
    print("Server error: \(msg)")
default:
    break
}
```

### Enum case patterns

```swift
enum Result {
    case success(Data)
    case failure(Error, retryable: Bool)
}

let result = Result.failure(SomeError(), retryable: true)

switch result {
case .success(let data):
    process(data)
case .failure(_, retryable: true):
    retry()
case .failure(let error, retryable: false):
    report(error)
}
```

### Optional pattern (x?)

```swift
let value: Int? = 42

switch value {
case let x?:       // .some(x)
    print(x)       // 42
case nil:          // .none
    print("nil")
}

// В коллекции
let scores: [Int?] = [nil, 85, nil, 92, 78]
for case let score? in scores {
    print(score)  // 85, 92, 78 (пропускает nil)
}
```

### Type-casting patterns

```swift
let items: [Any] = [42, "hello", 3.14, true]

for item in items {
    switch item {
    case let n as Int:
        print("Int: \(n)")
    case let s as String:
        print("String: \(s)")
    case is Double:
        print("It's a Double")
    default:
        print("Other")
    }
}
```

### Expression pattern (~=)

Оператор `~=` используется в `switch` для сопоставления:

```swift
// Range ~= содержит
switch 42 {
case 0..<10: print("small")
case 10..<100: print("medium")  // ✓
default: print("large")
}

// Это работает потому что:
// (0..<10) ~= 42  → false
// (10..<100) ~= 42 → true
```

### Where clause

```swift
let point = (1, -1)

switch point {
case let (x, y) where x == y:
    print("On the line x == y")
case let (x, y) where x == -y:
    print("On the line x == -y")  // ✓
case let (x, y):
    print("(\(x), \(y))")
}

// С enum
switch result {
case .success(let data) where data.count > 0:
    process(data)
case .success:
    print("Empty success")
case .failure:
    break
}
```

### Compound patterns (запятая)

```swift
switch character {
case "a", "e", "i", "o", "u":
    print("Vowel")
case "b", "c", "d", "f", "g":
    print("Consonant")
default:
    print("Other")
}

// С value binding
switch point {
case (0, 0), (1, 1):  // два конкретных значения
    print("Special point")
case let (x, y) where x == y:
    print("On diagonal")
default:
    break
}
```

## if case / guard case / for case

### if case

```swift
enum Delivery {
    case shipped(trackingNumber: String)
    case processing
    case delivered(date: Date)
}

let status = Delivery.shipped(trackingNumber: "ABC123")

if case .shipped(let tracking) = status {
    print("Shipped: \(tracking)")  // "Shipped: ABC123"
}

// С where
if case .shipped(let tracking) = status, tracking.hasPrefix("ABC") {
    print("ABC shipment")
}
```

### guard case

```swift
func handleDelivery(_ status: Delivery) {
    guard case .shipped(let tracking) = status else {
        print("Not shipped yet")
        return
    }
    // tracking доступен дальше
    print("Tracking: \(tracking)")
}
```

### for case

```swift
let responses: [Result] = [
    .success(data1),
    .failure(error1, retryable: true),
    .success(data2),
    .failure(error2, retryable: false)
]

// Только success
for case .success(let data) in responses {
    process(data)
}

// Optional filtering
let numbers: [Int?] = [1, nil, 3, nil, 5]
for case let n? in numbers {
    print(n)  // 1, 3, 5
}
```

## Кастомный pattern matching

Перегрузка оператора `~=` для кастомного matching:

```swift
// Regex matching в switch
struct RegexPattern {
    let pattern: String
}

func ~= (pattern: RegexPattern, value: String) -> Bool {
    value.range(of: pattern.pattern, options: .regularExpression) != nil
}

let email = "user@example.com"

switch email {
case RegexPattern(pattern: #"^\w+@\w+\.\w+$"#):
    print("Valid email")  // ✓
default:
    print("Invalid")
}

// Функция как паттерн
func ~= (predicate: (Int) -> Bool, value: Int) -> Bool {
    predicate(value)
}

func isEven(_ n: Int) -> Bool { n % 2 == 0 }
func isPositive(_ n: Int) -> Bool { n > 0 }

switch 42 {
case isEven:
    print("Even")  // ✓
case isPositive:
    print("Positive")
default:
    break
}
```

## Сравнение с JS

| Аспект | Swift | JavaScript |
|--------|-------|-----------|
| Exhaustiveness | Компилятор проверяет | Нет проверки |
| Fallthrough | Нет (explicit `fallthrough`) | Да (нужен `break`) |
| Ranges | `case 1...10:` | Нет (цепочка if) |
| Binding | `case let x:` | Нет |
| Tuple matching | `case (0, let y):` | Нет |
| Where clause | `case let x where x > 0:` | Нет |
| Optional pattern | `case let x?:` | Нет |
| Custom ~= | Да | Нет |
| Destructuring | `if case .enum(let v) = x` | `const { a, b } = obj` |

```swift
// Swift — мощный pattern matching
switch response {
case (200..<300, let body) where !body.isEmpty:
    process(body)
case (401, _):
    refreshToken()
case (let code, _) where code >= 500:
    reportServerError(code)
default:
    break
}

// JS — нет аналога, нужна цепочка if-else
// if (code >= 200 && code < 300 && body) { process(body) }
// else if (code === 401) { refreshToken() }
// else if (code >= 500) { reportServerError(code) }
```
