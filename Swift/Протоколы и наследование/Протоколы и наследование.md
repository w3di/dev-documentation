## Определение протоколов

Протокол определяет **контракт** — набор требований, которые должен реализовать подписывающийся тип:

```swift
protocol Drawable {
    var boundingBox: CGRect { get }          // property requirement
    func draw()                              // method requirement
    mutating func resize(by factor: Double)  // mutating для value types
}

struct Circle: Drawable {
    var radius: Double
    var center: CGPoint

    var boundingBox: CGRect {
        CGRect(x: center.x - radius, y: center.y - radius,
               width: radius * 2, height: radius * 2)
    }

    func draw() { print("Drawing circle r=\(radius)") }

    mutating func resize(by factor: Double) {
        radius *= factor
    }
}
```

### Property requirements

```swift
protocol Named {
    var name: String { get }       // read-only (можно реализовать как var или let)
    var id: String { get set }     // read-write (только var)
    static var typeName: String { get }  // static/class property
}
```

- `{ get }` — можно реализовать как stored, computed, или `let`
- `{ get set }` — только `var` (stored или computed с setter)

### Class-only протоколы

```swift
protocol Delegate: AnyObject {  // только классы могут подписаться
    func didUpdate()
}

class Controller {
    weak var delegate: Delegate?  // weak возможен только с AnyObject
}
```

### @objc optional

```swift
@objc protocol TableDelegate {
    @objc optional func didSelectRow(at index: Int)
    @objc optional func numberOfSections() -> Int
}

class MyDelegate: NSObject, TableDelegate {
    // Можно не реализовывать optional методы
}
```

## Protocol composition

Комбинирование нескольких протоколов через `&`:

```swift
protocol Named { var name: String { get } }
protocol Aged { var age: Int { get } }

func greet(_ person: Named & Aged) {
    print("Hello, \(person.name), age \(person.age)")
}

// Можно комбинировать с классом
func configure(_ view: UIView & Drawable) {
    view.draw()
    view.frame = view.boundingBox
}

// Typealias для удобства
typealias Identifiable = Named & Aged & Codable
```

## Protocol extensions

Добавляют **реализации по умолчанию** и вспомогательные методы:

```swift
protocol Container {
    associatedtype Item
    var count: Int { get }
    subscript(index: Int) -> Item { get }
}

extension Container {
    // Default implementation
    var isEmpty: Bool { count == 0 }

    var first: Item? {
        isEmpty ? nil : self[0]
    }

    var last: Item? {
        isEmpty ? nil : self[count - 1]
    }

    func allSatisfy(_ predicate: (Item) -> Bool) -> Bool {
        for i in 0..<count {
            if !predicate(self[i]) { return false }
        }
        return true
    }
}
```

### Dispatch ловушка

```swift
protocol Greetable {
    func greet() -> String        // requirement — witness table dispatch
}

extension Greetable {
    func greet() -> String {      // default implementation
        return "Hello"
    }

    func shout() -> String {      // НЕ requirement — static dispatch
        return greet().uppercased()
    }
}

struct Person: Greetable {
    func greet() -> String { "Hi, I'm a person" }
    func shout() -> String { "PERSON SHOUTING" }
}

let person = Person()
person.greet()  // "Hi, I'm a person" ✓
person.shout()  // "PERSON SHOUTING" ✓

let greetable: any Greetable = person
greetable.greet()  // "Hi, I'm a person" ✓ (witness table)
greetable.shout()  // "HELLO" ✗ (static dispatch → extension default!)
```

> **Правило:** Если метод должен быть полиморфным — объяви его в **требованиях** протокола, а не только в extension.

## Associated types

```swift
protocol Collection {
    associatedtype Element
    associatedtype Index: Comparable

    var startIndex: Index { get }
    var endIndex: Index { get }
    subscript(position: Index) -> Element { get }
}

// Конкретный тип определяет associated type
struct IntArray: Collection {
    typealias Element = Int  // можно опустить — компилятор выведет
    typealias Index = Int

    private var storage: [Int]

    var startIndex: Int { 0 }
    var endIndex: Int { storage.count }

    subscript(position: Int) -> Int {
        storage[position]
    }
}
```

### Primary associated types (Swift 5.7+)

```swift
protocol Collection<Element> {  // Element — primary associated type
    associatedtype Element
    associatedtype Index
    // ...
}

// Теперь можно указать конкретный тип в угловых скобках
func process(_ items: any Collection<String>) { /* ... */ }
func transform<C: Collection<Int>>(_ c: C) -> [Int] { /* ... */ }
```

### PAT (Protocol with Associated Type) — ограничения

```swift
protocol Stackable {
    associatedtype Item
    mutating func push(_ item: Item)
    func peek() -> Item?
}

// Нельзя использовать как тип напрямую без any/some (до Swift 5.7)
// var stack: Stackable  // Ошибка — PAT

// Решения:
// 1. any (existential)
var stack: any Stackable  // Swift 5.7+, теряет type info

// 2. some (opaque)
func makeStack() -> some Stackable { /* ... */ }

// 3. Generics
func process<S: Stackable>(stack: S) where S.Item: Equatable { /* ... */ }

// 4. Type erasure
struct AnyStack<T>: Stackable {
    private var _push: (T) -> Void
    private var _peek: () -> T?

    init<S: Stackable>(_ base: S) where S.Item == T {
        var base = base
        _push = { base.push($0) }
        _peek = { base.peek() }
    }

    mutating func push(_ item: T) { _push(item) }
    func peek() -> T? { _peek() }
}
```

## Existential vs Opaque types

### any — Existential type

```swift
// any Protocol — dynamic dispatch, runtime полиморфизм
var shapes: [any Drawable] = [Circle(), Square(), Triangle()]

// Стоимость: existential container (40 байт), возможна heap-аллокация
// Гибкость: можно хранить разные типы в одной коллекции
```

### some — Opaque type

```swift
// some Protocol — static dispatch, compile-time тип
func makeShape() -> some Drawable {
    Circle(radius: 5)  // конкретный тип скрыт, но компилятор его знает
}

// Стоимость: нулевая (прямой вызов)
// Ограничение: всегда возвращает один и тот же конкретный тип
```

### Сравнение

| | `any Protocol` | `some Protocol` | `<T: Protocol>` |
|--|----------------|----------------|----------------|
| Dispatch | Dynamic (witness table) | Static (direct) | Static (specialized) |
| Тип известен | В runtime | Компилятору | Компилятору |
| Разные типы | Да | Нет | Нет (per call) |
| Overhead | Existential container | Нет | Нет |
| Использование | Гетерогенные коллекции | Return types, properties | Параметры |

## Dispatch

### V-table (классы)

```swift
class Animal {
    func speak() { }  // Слот 0 в V-table
    func eat() { }    // Слот 1 в V-table
}

class Dog: Animal {
    override func speak() { }  // Заменяет слот 0
    func fetch() { }           // Слот 2
}

// V-table Dog: [Dog.speak, Animal.eat, Dog.fetch]
```

### Witness table (протоколы)

```swift
protocol Printable {
    func display()    // Слот 0 в witness table
    var title: String { get }  // Слот 1
}

struct Report: Printable {
    func display() { /* ... */ }
    var title: String { "Report" }
}

// Witness table Report: [Report.display, Report.title.getter]
```

При вызове через `any Printable` — runtime ищет witness table для конкретного типа и вызывает нужный слот.

## Protocol-Oriented Programming

POP — парадигма, предпочитающая протоколы и value types вместо наследования:

```swift
// Вместо базового класса — протокол + extension
protocol Vehicle {
    var speed: Double { get set }
    var fuel: Double { get set }
}

extension Vehicle {
    mutating func accelerate(by delta: Double) {
        speed += delta
        fuel -= delta * 0.1
    }

    var isMoving: Bool { speed > 0 }
}

// Composition вместо inheritance
protocol Electric {
    var batteryLevel: Double { get set }
}

protocol Autonomous {
    func navigate(to destination: String)
}

// Любой тип может подписаться на любую комбинацию
struct Tesla: Vehicle, Electric, Autonomous {
    var speed: Double = 0
    var fuel: Double = 0
    var batteryLevel: Double = 100

    func navigate(to destination: String) {
        print("Navigating to \(destination)")
    }
}
```

### Преимущества POP:
1. **Множественная подписка** — struct может реализовать много протоколов
2. **Value types** — нет shared mutable state
3. **Default implementations** — переиспользование кода без наследования
4. **Ретроактивная подписка** — можно подписать существующий тип

## Наследование классов

```swift
class Shape {
    var color: String

    init(color: String) { self.color = color }

    func area() -> Double { 0 }

    func describe() -> String {
        "\(color) shape with area \(area())"
    }
}

class Circle: Shape {
    var radius: Double

    init(color: String, radius: Double) {
        self.radius = radius
        super.init(color: color)  // вызов super обязателен
    }

    override func area() -> Double {
        .pi * radius * radius
    }
}

class FilledCircle: Circle {
    var fillOpacity: Double

    init(color: String, radius: Double, fillOpacity: Double) {
        self.fillOpacity = fillOpacity
        super.init(color: color, radius: radius)
    }

    override func describe() -> String {
        super.describe() + " (opacity: \(fillOpacity))"
    }
}
```

### final

```swift
final class Singleton {
    // Нельзя наследовать
}

class Base {
    final func validate() { }  // Нельзя переопределить
}
```

## Conditional и Retroactive conformance

### Conditional conformance

```swift
// Array<Element> соответствует Equatable ТОЛЬКО если Element: Equatable
extension Array: Equatable where Element: Equatable {
    // == автоматически синтезируется
}

[1, 2, 3] == [1, 2, 3]  // true — Int: Equatable

// Кастомный пример
struct Stack<T> {
    var items: [T] = []
}

extension Stack: Equatable where T: Equatable {}
extension Stack: Hashable where T: Hashable {}
extension Stack: Codable where T: Codable {}
```

### Retroactive conformance

Подписка типа из другой библиотеки на протокол:

```swift
// Подписываем чужой тип на свой протокол
extension URL: Identifiable {
    public var id: String { absoluteString }
}

// Swift 5.9+ — предупреждение, нужен @retroactive
extension URL: @retroactive Identifiable {
    public var id: String { absoluteString }
}
```

## Сравнение с JS/TS

| Аспект | Swift Protocol | TS Interface | JS Prototype |
|--------|---------------|-------------|-------------|
| Runtime проверка | `is`, `as?` | Нет (стирается) | `instanceof` |
| Default implementation | Extension | Нет | Prototype methods |
| Associated types | Да | Generics | Нет |
| Множественная подписка | Да | Да | Нет (1 prototype) |
| Conditional conformance | Да | Conditional types | Нет |
| Retroactive | Да (extension) | Declaration merging | Monkey-patching |
| Dispatch | Witness table | Стирается в JS | Hidden classes |
| Composition | `A & B` | `A & B` (intersection) | Mixins |

```swift
// Swift — протокол с associated type
protocol Repository {
    associatedtype Item: Identifiable
    func find(by id: Item.ID) -> Item?
    func save(_ item: Item) throws
}

// TS — interface с generic
// interface Repository<T extends Identifiable> {
//     find(id: T['id']): T | undefined
//     save(item: T): void
// }
```
