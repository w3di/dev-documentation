## Синтаксис

Extension добавляет функциональность к **существующему типу** (даже если у тебя нет исходного кода):

```swift
extension Int {
    var isEven: Bool { self % 2 == 0 }

    func repeated(_ action: (Int) -> Void) {
        for i in 0..<self { action(i) }
    }
}

print(42.isEven)     // true
3.repeated { print($0) }  // 0, 1, 2
```

## Computed properties

```swift
extension Double {
    var km: Double { self * 1_000 }
    var m: Double { self }
    var cm: Double { self / 100 }
    var mm: Double { self / 1_000 }
}

let marathon = 42.km + 195.m  // 42_195.0 метров

// ⚠️ НЕЛЬЗЯ добавлять stored properties
// extension Int {
//     var cache: [Int: String]  // Ошибка!
```

## Методы

```swift
extension String {
    func truncated(to length: Int, trailing: String = "...") -> String {
        if count <= length { return self }
        return String(prefix(length)) + trailing
    }

    // Mutating для value types
    mutating func trimWhitespace() {
        self = trimmingCharacters(in: .whitespaces)
    }
}

"Hello, World!".truncated(to: 5)  // "Hello..."
```

## Инициализаторы

```swift
// Для struct — сохраняем memberwise init
struct Size {
    var width: Double
    var height: Double
}

extension Size {
    init(square side: Double) {
        self.init(width: side, height: side)  // memberwise сохранён!
    }
}

let s1 = Size(width: 10, height: 20)  // memberwise — работает
let s2 = Size(square: 15)              // кастомный — тоже

// Для class — только convenience
extension UIColor {
    convenience init(hex: String) {
        // парсинг hex → RGB
        self.init(red: r, green: g, blue: b, alpha: 1.0)
    }
}
```

## Subscripts

```swift
extension Array {
    subscript(safe index: Int) -> Element? {
        indices.contains(index) ? self[index] : nil
    }
}

let arr = [1, 2, 3]
print(arr[safe: 5])  // nil (вместо crash)
```

## Nested types

```swift
extension Int {
    enum Kind {
        case negative, zero, positive
    }

    var kind: Kind {
        switch self {
        case 0: return .zero
        case let x where x > 0: return .positive
        default: return .negative
        }
    }
}

print(42.kind)   // .positive
print((-1).kind) // .negative
```

## Protocol conformance

```swift
// Подписка через extension
struct Vector {
    var x: Double
    var y: Double
}

extension Vector: CustomStringConvertible {
    var description: String { "(\(x), \(y))" }
}

extension Vector: Equatable {
    // Автосинтез == (все свойства Equatable)
}

extension Vector: Hashable {
    // Автосинтез hash(into:)
}

print(Vector(x: 1, y: 2))  // "(1.0, 2.0)"
```

### @retroactive

```swift
// Подписка чужого типа на чужой протокол
extension URL: @retroactive Identifiable {
    public var id: String { absoluteString }
}
```

## Conditional extensions

Расширения с ограничениями через `where`:

```swift
// Только для Array<Element> где Element: Numeric
extension Array where Element: Numeric {
    var sum: Element {
        reduce(0, +)
    }

    var average: Double where Element: BinaryInteger {
        isEmpty ? 0 : Double(sum) / Double(count)
    }
}

[1, 2, 3, 4].sum      // 10
[1, 2, 3, 4].average   // 2.5
// ["a", "b"].sum      // Ошибка! String не Numeric

// Для Optional
extension Optional where Wrapped: Collection {
    var isNilOrEmpty: Bool {
        self?.isEmpty ?? true
    }
}

let names: [String]? = nil
print(names.isNilOrEmpty)  // true

// Conditional conformance
extension Array: Encodable where Element: Encodable {}
extension Optional: Equatable where Wrapped: Equatable {}
```

## Dispatch

### Static dispatch (default)

Методы в extensions используют **static dispatch** по умолчанию:

```swift
class Animal {
    func speak() { print("...") }
}

extension Animal {
    func describe() { print("I'm an animal") }
}

class Dog: Animal {
    override func speak() { print("Woof") }  // V-table — dynamic

    func describe() { print("I'm a dog") }   // shadow, NOT override
}

let animal: Animal = Dog()
animal.speak()     // "Woof"      — dynamic dispatch (V-table)
animal.describe()  // "I'm an animal" — static dispatch (тип Animal)

let dog: Dog = Dog()
dog.describe()     // "I'm a dog"  — static dispatch (тип Dog)
```

### Когда extension method использует dynamic dispatch:
1. Метод объявлен в протоколе как **requirement** и реализован в extension
2. Метод помечен `@objc` (Objective-C runtime)

```swift
protocol Greetable {
    func greet()  // requirement → dynamic через witness table
}

extension Greetable {
    func greet() { print("Hello") }         // default — dynamic (requirement)
    func farewell() { print("Goodbye") }    // НЕ requirement — static
}
```

## Организация кода

Распространённый паттерн — разделение conformances по extensions:

```swift
// MARK: - Core
struct UserProfile {
    var id: UUID
    var name: String
    var email: String
}

// MARK: - Equatable
extension UserProfile: Equatable {}

// MARK: - Hashable
extension UserProfile: Hashable {}

// MARK: - Codable
extension UserProfile: Codable {
    enum CodingKeys: String, CodingKey {
        case id, name, email
    }
}

// MARK: - CustomStringConvertible
extension UserProfile: CustomStringConvertible {
    var description: String { "\(name) <\(email)>" }
}

// MARK: - Helpers
extension UserProfile {
    var displayName: String {
        name.isEmpty ? email : name
    }

    func isValid() -> Bool {
        !name.isEmpty && email.contains("@")
    }
}
```

## Ограничения

1. **Нельзя** добавлять stored properties
2. **Нельзя** добавлять designated initializers для class
3. **Нельзя** переопределять существующие методы (только shadow)
4. **Нельзя** добавлять deinit

## Сравнение с JS

| Аспект | Swift Extension | JS Prototype | JS Monkey-patching |
|--------|----------------|-------------|-------------------|
| Безопасность | Compile-time проверки | Runtime errors | Runtime errors |
| Stored properties | Нет | Да | Да |
| Namespace collision | Ошибка компилятора | Перезапись | Перезапись |
| Conditional | `where` clause | Нет | Нет |
| Retroactive conformance | Да | Да (prototype) | Да |
| Рекомендуется | Да (идиоматично) | Осторожно | Не рекомендуется |

```swift
// Swift — безопасное расширение
extension Array where Element: Comparable {
    func isSorted() -> Bool {
        zip(self, dropFirst()).allSatisfy { $0 <= $1 }
    }
}

// JS — monkey-patching (опасно)
// Array.prototype.isSorted = function() {
//     return this.every((v, i) => i === 0 || this[i-1] <= v);
// }
```
