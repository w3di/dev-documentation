## Struct vs Class

| Свойство | `struct` | `class` |
|----------|---------|---------|
| Тип | Value type | Reference type |
| Хранение | Stack (обычно) | Heap |
| Копирование | Копия значения (COW) | Копия ссылки |
| Наследование | Нет | Да |
| deinit | Нет | Да |
| Reference counting | Нет | Да (ARC) |
| Mutability | `mutating` для методов | Свободная мутация |
| Identity (`===`) | Нет | Да |

```swift
// Struct — значимый тип
struct Point {
    var x: Double
    var y: Double
}

var p1 = Point(x: 0, y: 0)
var p2 = p1      // копия
p2.x = 10
print(p1.x)      // 0 — не изменился

// Class — ссылочный тип
class Circle {
    var radius: Double
    init(radius: Double) { self.radius = radius }
}

let c1 = Circle(radius: 5)
let c2 = c1      // ссылка на тот же объект
c2.radius = 10
print(c1.radius)  // 10 — изменился!
```

## Инициализаторы

### Memberwise initializer (только struct)

```swift
struct Size {
    var width: Double
    var height: Double
    // Компилятор автоматически создаёт:
    // init(width: Double, height: Double)
}

let size = Size(width: 100, height: 50)

// Если добавить кастомный init — memberwise пропадает
// Решение: вынести кастомный init в extension
extension Size {
    init(square side: Double) {
        self.init(width: side, height: side)
    }
}
```

### Designated и Convenience (class)

```swift
class Vehicle {
    var name: String
    var speed: Double

    // Designated — основной, инициализирует все свойства
    init(name: String, speed: Double) {
        self.name = name
        self.speed = speed
    }

    // Convenience — вспомогательный, вызывает designated
    convenience init(name: String) {
        self.init(name: name, speed: 0)
    }
}

class Car: Vehicle {
    var doors: Int

    // Designated подкласса ДОЛЖЕН вызвать designated суперкласса
    init(name: String, speed: Double, doors: Int) {
        self.doors = doors
        super.init(name: name, speed: speed)  // обязательно
    }

    // Convenience подкласса вызывает designated ЭТОГО ЖЕ класса
    convenience init(sedan name: String) {
        self.init(name: name, speed: 0, doors: 4)
    }
}
```

### Правила инициализации классов (двухфазная)

**Фаза 1:** Каждое stored property получает начальное значение (снизу вверх по иерархии)
**Фаза 2:** Каждый класс может дополнительно настроить свои свойства (сверху вниз)

```swift
class Base {
    var x: Int
    init(x: Int) {
        self.x = x        // Фаза 1: инициализация x
        // Фаза 2: можно использовать self
        print("Base ready")
    }
}

class Sub: Base {
    var y: Int
    init(x: Int, y: Int) {
        self.y = y          // Фаза 1: сначала свои свойства
        super.init(x: x)   // Фаза 1: затем super
        // Фаза 2: self полностью инициализирован
        self.x = x + y     // можно менять унаследованные свойства
    }
}
```

### Required initializer

```swift
class Document {
    var title: String

    required init(title: String) {
        self.title = title
    }
}

class Report: Document {
    var pages: Int = 0

    // ОБЯЗАНЫ реализовать required init
    required init(title: String) {
        super.init(title: title)
    }
}
```

### Failable initializer

```swift
struct Temperature {
    var celsius: Double

    init?(fahrenheit: Double) {
        let celsius = (fahrenheit - 32) * 5 / 9
        guard celsius >= -273.15 else { return nil }  // абсолютный ноль
        self.celsius = celsius
    }
}

let temp = Temperature(fahrenheit: -500)  // nil
let valid = Temperature(fahrenheit: 72)   // Temperature(celsius: 22.22)
```

## Наследование

Только для классов:

```swift
class Animal {
    var name: String

    init(name: String) { self.name = name }

    func speak() -> String {
        return "..."
    }
}

class Dog: Animal {
    override func speak() -> String {
        return "Woof!"
    }
}

class GuideDog: Dog {
    override func speak() -> String {
        return super.speak() + " (quietly)"
    }
}
```

### final

```swift
final class Singleton {
    // Нельзя наследовать от final класса
    static let shared = Singleton()
    private init() {}
}

class Base {
    final func criticalMethod() {
        // Нельзя переопределить в подклассе
    }
}
```

`final` также даёт **оптимизацию** — компилятор использует **direct dispatch** вместо V-table.

## Методы

### Mutating (struct/enum)

```swift
struct Counter {
    var count = 0

    mutating func increment() {
        count += 1
    }

    mutating func reset() {
        self = Counter()  // можно заменить self целиком
    }
}

var counter = Counter()
counter.increment()
print(counter.count)  // 1

// let constant = Counter()
// constant.increment()  // Ошибка! let struct не может вызывать mutating
```

### Static и class методы

```swift
struct MathHelper {
    static func square(_ x: Double) -> Double {
        return x * x
    }
}

class Vehicle {
    class func defaultSpeed() -> Double {  // class — можно override
        return 0
    }

    static func maxSpeed() -> Double {     // static — нельзя override
        return 300
    }
}

class Car: Vehicle {
    override class func defaultSpeed() -> Double { 60 }
    // override static func maxSpeed()  // Ошибка!
}
```

## Subscripts

```swift
struct Matrix {
    let rows: Int, columns: Int
    var grid: [Double]

    init(rows: Int, columns: Int) {
        self.rows = rows
        self.columns = columns
        grid = Array(repeating: 0.0, count: rows * columns)
    }

    subscript(row: Int, column: Int) -> Double {
        get {
            precondition(row >= 0 && row < rows && column >= 0 && column < columns)
            return grid[row * columns + column]
        }
        set {
            precondition(row >= 0 && row < rows && column >= 0 && column < columns)
            grid[row * columns + column] = newValue
        }
    }
}

var matrix = Matrix(rows: 2, columns: 2)
matrix[0, 0] = 1.0
matrix[1, 1] = 1.0
print(matrix[0, 0])  // 1.0

// Static subscript
enum Settings {
    static var storage: [String: Any] = [:]

    static subscript(key: String) -> Any? {
        get { storage[key] }
        set { storage[key] = newValue }
    }
}
Settings["theme"] = "dark"
```

## Dispatch механизмы

Swift использует три типа диспетчеризации:

### 1. Direct (Static) Dispatch

Адрес функции известен **на этапе компиляции**. Самый быстрый.

```swift
// Direct dispatch:
// - struct/enum методы
// - final методы класса
// - private методы класса
// - static/class final методы
// - extension методы (без @objc)

struct Point {
    func distance(to other: Point) -> Double { /* direct */ }
}

final class Config {
    func load() { /* direct */ }
}
```

### 2. V-Table (Virtual Table) Dispatch

Таблица указателей на функции, определяется в runtime по типу объекта:

```swift
// V-table dispatch:
// - class методы (не final, не private)

class Shape {
    func area() -> Double { 0 }   // через V-table
}

class Circle: Shape {
    var radius: Double = 0
    override func area() -> Double { .pi * radius * radius }
}

let shape: Shape = Circle()
shape.area()  // V-table lookup → Circle.area()
```

### 3. Witness Table (Protocol) Dispatch

Для протоколов — таблица, маппящая требования протокола на конкретные реализации:

```swift
protocol Drawable {
    func draw()  // через witness table
}

extension Drawable {
    func describe() { }  // через direct dispatch!
}

struct Square: Drawable {
    func draw() { print("Square") }
    func describe() { print("I'm a square") }
}

let d: any Drawable = Square()
d.draw()      // witness table → Square.draw() ✓
d.describe()  // direct dispatch → Drawable.describe() (НЕ Square.describe()!)
```

> **Важно:** Методы, объявленные в `extension` протокола (но не в требованиях) — используют **static dispatch**. Это частый источник багов.

## Stack vs Heap

```swift
// Stack (быстро, нет ARC)
struct SmallData {
    var a: Int
    var b: Int
}

// Heap (медленнее, ARC overhead)
class BigObject {
    var data: [Int] = []
}

// Struct с классом внутри — смешанная аллокация
struct Container {
    var value: Int        // на стеке
    var ref: BigObject    // указатель на стеке, объект на heap
    // ARC применяется к ref!
}
```

### Когда struct попадает на heap

1. **Existential container** — если значение > 3 words (24 байт)
2. **Захват замыканием** — захваченные переменные уходят на heap
3. **Протокольный тип** — `any Protocol` использует existential

## Когда что использовать

### Используй struct (по умолчанию):
- Данные без identity (координаты, размеры, конфигурация)
- Thread safety без синхронизации
- Предсказуемое поведение при копировании
- Apple рекомендует struct по умолчанию

### Используй class:
- Нужна **identity** (`===`)
- Нужно **наследование**
- Нужен **shared mutable state**
- Нужен **deinit** для cleanup ресурсов
- Interop с Objective-C

```swift
// ✅ struct — данные
struct UserProfile {
    var name: String
    var email: String
    var avatarURL: URL?
}

// ✅ class — identity, shared state, ресурсы
class NetworkSession {
    private var urlSession: URLSession
    weak var delegate: SessionDelegate?

    deinit { urlSession.invalidateAndCancel() }
}
```

## Сравнение с JS

| Аспект | Swift struct | Swift class | JS object | JS class |
|--------|-------------|------------|-----------|----------|
| Тип | Value | Reference | Reference | Reference |
| Копирование | Глубокая копия (COW) | Копия ссылки | Копия ссылки | Копия ссылки |
| Наследование | Нет | Классическое | Прототипное | Прототипное (sugar) |
| Init | Memberwise (авто) | Designated + convenience | Нет | `constructor()` |
| Мутабельность | `mutating` для let | Свободная (let = const ref) | Свободная | Свободная |
| Dispatch | Direct | V-table | V8 hidden classes + IC | V8 hidden classes + IC |
| Деструктор | Нет | `deinit` | Нет (`FinalizationRegistry`) | Нет |

```swift
// Swift: let struct = полная иммутабельность
let point = Point(x: 1, y: 2)
// point.x = 3  // Ошибка!

// JS: const object = только ссылка const
// const obj = { x: 1 }; obj.x = 3; // OK!

// Swift: let class = ссылка const, содержимое мутабельно
let circle = Circle(radius: 5)
circle.radius = 10  // OK — меняем содержимое, не ссылку
```
