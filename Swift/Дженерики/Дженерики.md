## Generic functions

```swift
func swapValues<T>(_ a: inout T, _ b: inout T) {
    let temp = a
    a = b
    b = temp
}

var x = 5, y = 10
swapValues(&x, &y)  // x=10, y=5

var a = "hello", b = "world"
swapValues(&a, &b)  // a="world", b="hello"

// Множественные type parameters
func zip<A, B>(_ a: [A], _ b: [B]) -> [(A, B)] {
    Array(Swift.zip(a, b))
}
```

## Generic types

```swift
struct Stack<Element> {
    private var items: [Element] = []

    mutating func push(_ item: Element) {
        items.append(item)
    }

    mutating func pop() -> Element? {
        items.popLast()
    }

    var top: Element? { items.last }
    var isEmpty: Bool { items.isEmpty }
    var count: Int { items.count }
}

var intStack = Stack<Int>()
intStack.push(1)
intStack.push(2)
print(intStack.pop())  // Optional(2)

var stringStack = Stack<String>()
stringStack.push("hello")

// Generic enum
enum Result<Success, Failure: Error> {
    case success(Success)
    case failure(Failure)
}

// Generic class
class Cache<Key: Hashable, Value> {
    private var storage: [Key: Value] = [:]

    func get(_ key: Key) -> Value? { storage[key] }
    func set(_ key: Key, value: Value) { storage[key] = value }
}
```

## Type constraints

### Протокольные ограничения

```swift
func findIndex<T: Equatable>(of value: T, in array: [T]) -> Int? {
    array.firstIndex(of: value)
}

// Множественные ограничения
func process<T: Hashable & Codable>(_ value: T) { /* ... */ }
```

### Where clause

```swift
func allEqual<C: Collection>(_ collection: C) -> Bool
    where C.Element: Equatable
{
    guard let first = collection.first else { return true }
    return collection.allSatisfy { $0 == first }
}

// Сложные ограничения
func merge<C1: Collection, C2: Collection>(_ a: C1, _ b: C2) -> [C1.Element]
    where C1.Element == C2.Element, C1.Element: Comparable
{
    (Array(a) + Array(b)).sorted()
}
```

### Generic subscripts

```swift
extension Dictionary {
    subscript<T>(key: Key, default defaultValue: @autoclosure () -> T) -> T
        where Value == T?
    {
        self[key] ?? defaultValue()
    }
}
```

## Associated types

```swift
protocol Container {
    associatedtype Item
    var count: Int { get }
    mutating func append(_ item: Item)
    subscript(i: Int) -> Item { get }
}

// Primary associated types (Swift 5.7+)
protocol Container<Item> {
    associatedtype Item
    // ...
}

// Использование с primary associated types
func sum(_ container: some Container<Int>) -> Int {
    var total = 0
    for i in 0..<container.count {
        total += container[i]
    }
    return total
}
```

### Ограничения на associated types

```swift
protocol SortedContainer {
    associatedtype Item: Comparable  // Item должен быть Comparable

    var sorted: [Item] { get }
}

// Associated type с where clause
protocol Sequence {
    associatedtype Iterator: IteratorProtocol
    associatedtype Element where Element == Iterator.Element
}
```

## some vs any

### some — Opaque type (Swift 5.1+)

```swift
// Компилятор знает конкретный тип, вызывающий — нет
func makeCollection() -> some Collection {
    [1, 2, 3]  // всегда Array<Int>
}

// some в параметрах (Swift 5.7+) — сахар для generics
func process(_ value: some Equatable) { /* ... */ }
// эквивалент: func process<T: Equatable>(_ value: T) { /* ... */ }

// ✅ Direct dispatch, zero overhead
// ❌ Всегда один конкретный тип
```

### any — Existential type (Swift 5.6+)

```swift
// Runtime полиморфизм, разные типы
var shapes: [any Drawable] = [Circle(), Square()]

func draw(_ shape: any Drawable) {
    shape.draw()  // witness table dispatch
}

// ✅ Гетерогенные коллекции
// ❌ Overhead: existential container, dynamic dispatch
```

### Когда что

```swift
// some — когда тип один и тот же (return type, property)
var body: some View { Text("Hello") }

// any — когда типы разные (heterogeneous collections)
var delegates: [any Delegate] = []

// Generic <T> — когда вызывающий определяет тип
func decode<T: Decodable>(_ data: Data) throws -> T
```

## Conditional conformance

```swift
// Array<Element> Equatable только если Element: Equatable
extension Array: Equatable where Element: Equatable {}

// Optional<Wrapped> Codable только если Wrapped: Codable
extension Optional: Codable where Wrapped: Codable {}

// Кастомный пример
struct Pair<A, B> {
    var first: A
    var second: B
}

extension Pair: Equatable where A: Equatable, B: Equatable {}
extension Pair: Hashable where A: Hashable, B: Hashable {}
extension Pair: Codable where A: Codable, B: Codable {}

extension Pair: CustomStringConvertible where A: CustomStringConvertible, B: CustomStringConvertible {
    var description: String { "(\(first), \(second))" }
}
```

## Type erasure

Оборачивает конкретный тип в тип без generic параметров:

```swift
// Проблема: нельзя хранить разные Iterator в одной переменной
// let iter: IteratorProtocol  // Ошибка — PAT

// Решение: AnyIterator
struct AnyIterator<Element>: IteratorProtocol {
    private var _next: () -> Element?

    init<I: IteratorProtocol>(_ base: I) where I.Element == Element {
        var base = base
        _next = { base.next() }
    }

    mutating func next() -> Element? { _next() }
}

// Использование
let arrays: [[Int]] = [[1, 2], [3, 4, 5]]
var iterators: [AnyIterator<Int>] = arrays.map { AnyIterator($0.makeIterator()) }
```

### С existential (Swift 5.7+)

```swift
// any Collection<Int> — встроенный type erasure
var collections: [any Collection<Int>] = [[1, 2], Set([3, 4])]
```

## Phantom types

Типовые параметры, которые не используются в stored properties — для compile-time safety:

```swift
enum Validated {}
enum Unvalidated {}

struct Email<Status> {
    let address: String
}

func validate(_ email: Email<Unvalidated>) -> Email<Validated>? {
    guard email.address.contains("@") else { return nil }
    return Email<Validated>(address: email.address)
}

func send(to email: Email<Validated>) {
    print("Sending to \(email.address)")
}

let raw = Email<Unvalidated>(address: "user@example.com")
// send(to: raw)  // Ошибка компиляции! Не Validated

if let valid = validate(raw) {
    send(to: valid)  // OK
}
```

## Result builders

`@resultBuilder` — DSL для декларативного построения значений:

```swift
@resultBuilder
struct ArrayBuilder<Element> {
    static func buildBlock(_ components: Element...) -> [Element] {
        components
    }

    static func buildOptional(_ component: [Element]?) -> [Element] {
        component ?? []
    }

    static func buildEither(first component: [Element]) -> [Element] {
        component
    }

    static func buildEither(second component: [Element]) -> [Element] {
        component
    }

    static func buildArray(_ components: [[Element]]) -> [Element] {
        components.flatMap { $0 }
    }
}

func buildList<T>(@ArrayBuilder<T> _ content: () -> [T]) -> [T] {
    content()
}

let list = buildList {
    1
    2
    3
}
// [1, 2, 3]
```

## Parameter packs (Swift 5.9+)

Variadic generics — для работы с произвольным количеством типовых параметров:

```swift
// each — parameter pack
func tuple<each T>(_ value: repeat each T) -> (repeat each T) {
    (repeat each value)
}

let result = tuple(1, "hello", true)  // (Int, String, Bool)

// Практический пример: zip нескольких массивов
func zipAll<each S: Sequence>(
    _ sequence: repeat each S
) -> [(repeat (each S).Element)] {
    // ...
}

// Constraints на каждый элемент pack
func allEqual<each T: Equatable>(_ value: repeat each T, to other: repeat each T) -> Bool {
    // Сравнение каждой пары
    var result = true
    repeat result = result && (each value == each other)
    return result
}
```

## Специализация

Компилятор Swift **специализирует** generic функции для конкретных типов:

```swift
// Generic версия
func max<T: Comparable>(_ a: T, _ b: T) -> T {
    a >= b ? a : b
}

// Компилятор генерирует специализированные версии:
// func max_Int(_ a: Int, _ b: Int) -> Int { a >= b ? a : b }
// func max_String(_ a: String, _ b: String) -> String { ... }

// @_specialize — подсказка компилятору
@_specialize(where T == Int)
@_specialize(where T == Double)
func optimizedMax<T: Comparable>(_ a: T, _ b: T) -> T {
    a >= b ? a : b
}
```

Специализация устраняет:
- Witness table dispatch → direct dispatch
- Existential container → direct value
- Indirect access → inline access

Для кросс-модульной специализации нужен `@inlinable`.

## Сравнение с TS

| Аспект | Swift Generics | TypeScript Generics |
|--------|---------------|-------------------|
| Runtime | Реальные типы (specialized) | Стираются (type erasure) |
| Constraints | `T: Protocol`, `where` | `T extends Interface` |
| Associated types | Да | Нет (mapped types) |
| Conditional conformance | Да | Conditional types |
| Variadic | Parameter packs | Нет (variadic tuple types) |
| Opaque types | `some Protocol` | Нет |
| Default type | Нет | `T = DefaultType` |
| Специализация | Компилятором | Нет (стирание) |

```swift
// Swift
func decode<T: Decodable>(_ data: Data) throws -> T {
    try JSONDecoder().decode(T.self, from: data)
}

// TypeScript
// function decode<T>(data: string): T {
//     return JSON.parse(data) as T;  // unsafe cast, no real type info
// }
```
