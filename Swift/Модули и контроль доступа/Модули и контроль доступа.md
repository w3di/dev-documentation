## Модули

**Модуль** — единица распространения кода (framework, app, package). Импортируется целиком:

```swift
import Foundation    // весь модуль
import struct Foundation.URL  // только конкретный тип
import func Darwin.sqrt       // только функция

// Каждый Swift target (app, framework, library) — один модуль
// Всё внутри модуля видит друг друга (internal по умолчанию)
```

### import варианты

```swift
import ModuleName           // весь модуль
import struct Module.Type   // один struct
import class Module.Type    // один class
import enum Module.Type     // один enum
import protocol Module.Type // один protocol
import func Module.name     // одна функция
import var Module.name      // одна переменная
import typealias Module.Name  // один typealias
```

## Уровни доступа

5 уровней, от самого открытого к самому закрытому:

| Уровень | Видимость | Наследование/Override |
|---------|-----------|----------------------|
| `open` | Везде (и за пределами модуля) | Можно наследовать и override |
| `public` | Везде (и за пределами модуля) | Нельзя наследовать/override извне |
| `internal` | Внутри модуля | Да |
| `fileprivate` | Внутри файла | Да (в пределах файла) |
| `private` | Внутри объявления (+ extensions в том же файле) | Нет |

### open vs public

```swift
// В модуле MyFramework:
open class BaseView {
    open func draw() { }       // можно override в другом модуле
    public func validate() { } // НЕ можно override в другом модуле
}

public class Config {
    // Нельзя наследовать в другом модуле
    public var value: Int = 0
}

// В app (другой модуль):
class CustomView: BaseView {     // OK — BaseView is open
    override func draw() { }     // OK — draw() is open
    // override func validate() { }  // Ошибка! validate() is public, not open
}

// class MyConfig: Config { }    // Ошибка! Config is public, not open
```

### internal (по умолчанию)

```swift
// Без модификатора = internal
class MyService {        // internal
    var name: String     // internal
    func process() { }   // internal
}

// Видно внутри модуля, невидимо снаружи
```

### fileprivate

```swift
// file: Helpers.swift

fileprivate struct Helper {
    fileprivate func doWork() { print("Working") }
}

struct PublicAPI {
    func run() {
        let helper = Helper()  // OK — тот же файл
        helper.doWork()        // OK — тот же файл
    }
}

// В другом файле:
// let h = Helper()  // Ошибка! fileprivate
```

### private

```swift
class Account {
    private var balance: Double = 0  // видно только внутри Account

    private func validateAmount(_ amount: Double) -> Bool {
        amount > 0 && amount <= balance
    }

    func withdraw(_ amount: Double) -> Bool {
        guard validateAmount(amount) else { return false }
        balance -= amount
        return true
    }
}

// Extensions в ТОМ ЖЕ файле видят private
extension Account {
    func deposit(_ amount: Double) {
        balance += amount  // OK — extension в том же файле
    }
}

// Extensions в ДРУГОМ файле НЕ видят private
```

## Правила по умолчанию

### Тип ограничивает своих членов

```swift
// private тип → все члены фактически private
private struct Secret {
    var value: String      // фактически private (тип private)
    public var name: String  // Предупреждение: public бессмысленно
}

// internal тип → члены максимум internal
struct Config {          // internal
    var value: Int       // internal (максимум для internal типа)
    public var name: String  // Предупреждение
}

// public тип → члены internal по умолчанию (НЕ public!)
public struct PublicConfig {
    var value: Int            // internal! Не видно за пределами модуля
    public var name: String   // public — явно
}
```

### Инициализаторы

```swift
public struct User {
    public var name: String
    public var age: Int

    // Memberwise init — internal! Не видим за пределами модуля
    // User(name: "Alice", age: 30)  — только внутри модуля

    // Нужен явный public init
    public init(name: String, age: Int) {
        self.name = name
        self.age = age
    }
}
```

### Enum cases

```swift
public enum Direction {
    case north   // автоматически public (enum cases наследуют уровень enum)
    case south
}
```

### Tuple, Function types

Уровень доступа определяется **самым ограниченным** компонентом:

```swift
private struct A {}
internal struct B {}

// Тип (A, B) — фактически private (ограничен private A)
// func process() -> (A, B) — максимум private
```

## @testable import

```swift
// В тестах:
@testable import MyApp

// Позволяет обращаться к internal сущностям
// (но НЕ к private/fileprivate)

class MyTests: XCTestCase {
    func testInternalLogic() {
        let service = InternalService()  // internal → доступен с @testable
        XCTAssertTrue(service.validate())
    }
}
```

## @_exported import

Реэкспорт модуля — импортирующий получает доступ без отдельного import:

```swift
// В модуле MyFramework:
@_exported import Foundation

// Теперь любой, кто делает import MyFramework,
// автоматически получает Foundation

// ⚠️ Underscore-prefix = неофициальный API, может измениться
```

## Организация кода

### Один тип — один файл (или группа файлов через extensions)

```swift
// User.swift
public struct User {
    public let id: UUID
    public var name: String
    private var passwordHash: String
}

// User+Codable.swift
extension User: Codable {
    enum CodingKeys: String, CodingKey {
        case id, name
        // passwordHash не кодируется
    }
}

// User+Validation.swift
extension User {
    func isValid() -> Bool {
        !name.isEmpty && !passwordHash.isEmpty
    }
}
```

### Access control паттерны

```swift
// Паттерн: public interface, private implementation
public class NetworkClient {
    // Public API
    public func fetch(url: URL) async throws -> Data {
        let request = buildRequest(for: url)
        return try await execute(request)
    }

    // Private implementation
    private func buildRequest(for url: URL) -> URLRequest {
        var request = URLRequest(url: url)
        request.addValue(apiKey, forHTTPHeaderField: "Authorization")
        return request
    }

    private func execute(_ request: URLRequest) async throws -> Data {
        let (data, _) = try await session.data(for: request)
        return data
    }

    private let apiKey: String
    private let session: URLSession
}

// Паттерн: public read, private write
public struct Counter {
    public private(set) var count: Int = 0  // чтение — public, запись — private

    public mutating func increment() {
        count += 1
    }
}
```

## Сравнение с JS

| Аспект | Swift | JavaScript |
|--------|-------|-----------|
| Система модулей | Compile-time, один модуль = один target | ESM (`import`/`export`), CommonJS (`require`) |
| Уровни доступа | 5 (`open`/`public`/`internal`/`fileprivate`/`private`) | 2 (exported / not exported) |
| По умолчанию | `internal` (видно в модуле) | Не экспортировано |
| Export | `public`/`open` | `export` / `export default` |
| Private | `private` (compile-time enforced) | `#private` fields (runtime) |
| Circular deps | Нет проблем (один модуль) | Возможны (ESM hoisting помогает) |
| Testing | `@testable import` | Обычный import (нет контроля) |
| Dynamic import | Нет | `import()` |
| Tree shaking | Компилятор убирает dead code | Bundler (webpack, rollup) |

```swift
// Swift — granular access control
public class API {
    public func fetchUsers() { }       // видно извне модуля
    internal func buildQuery() { }     // только внутри модуля
    fileprivate var cache: [String] = []  // только в этом файле
    private var apiKey = "secret"      // только в этом объявлении
}

// JS — только exported или not
// export class API {
//     fetchUsers() { }      // доступно при import
//     #apiKey = "secret"    // private field (runtime check)
//     // нет fileprivate, internal, open
// }
```
