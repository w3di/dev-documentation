## is

Проверяет, является ли экземпляр определённым типом:

```swift
class Animal {}
class Dog: Animal {}
class Cat: Animal {}

let pet: Animal = Dog()

pet is Dog     // true
pet is Cat     // false
pet is Animal  // true

// В switch
func describe(_ animal: Animal) {
    switch animal {
    case is Dog: print("Dog")
    case is Cat: print("Cat")
    default: print("Unknown animal")
    }
}

// С протоколами
let value: Any = 42
value is CustomStringConvertible  // true (Int подписан)
```

## as

### as — upcasting (гарантированно успешный)

```swift
let dog = Dog()
let animal: Animal = dog as Animal  // upcasting — всегда безопасен
// Обычно as не нужен — upcasting происходит неявно
let animal2: Animal = dog  // эквивалент
```

### as? — conditional downcasting

```swift
let animal: Animal = Dog()

if let dog = animal as? Dog {
    print("It's a dog")  // ✓
}

if let cat = animal as? Cat {
    print("It's a cat")  // не выполнится
}

// С протоколами
let value: Any = "Hello"
if let str = value as? String {
    print(str.uppercased())  // "HELLO"
}
```

### as! — forced downcasting

```swift
let animal: Animal = Dog()
let dog = animal as! Dog   // OK
// let cat = animal as! Cat  // Runtime crash!

// Используй только когда 100% уверен в типе
```

### Type casting в коллекциях

```swift
let animals: [Animal] = [Dog(), Cat(), Dog(), Cat()]

// Фильтрация по типу
let dogs = animals.compactMap { $0 as? Dog }  // [Dog]

// for case
for case let dog as Dog in animals {
    print("Found a dog")
}
```

## Any и AnyObject

### Any — любой тип (value и reference)

```swift
var things: [Any] = []
things.append(42)
things.append(3.14)
things.append("hello")
things.append(Dog())
things.append((3, 5))
things.append({ (x: Int) in x * 2 })

for thing in things {
    switch thing {
    case let i as Int:
        print("Int: \(i)")
    case let d as Double:
        print("Double: \(d)")
    case let s as String:
        print("String: \(s)")
    case let f as (Int) -> Int:
        print("Function result: \(f(5))")
    default:
        print("Something else")
    }
}
```

### AnyObject — только reference types (классы)

```swift
let objects: [AnyObject] = [Dog(), NSString("hello")]
// let objects2: [AnyObject] = [42]  // Ошибка! Int — value type

// AnyObject нужен для:
// 1. Interop с Objective-C
// 2. Class-only протоколы
// 3. weak references
```

## Metatypes

### Type и .self

```swift
// Тип типа — метатип
let intType: Int.Type = Int.self
let stringType: String.Type = String.self

// type(of:) — динамический тип
let animal: Animal = Dog()
type(of: animal)  // Dog.Type (не Animal.Type!)
type(of: animal) == Dog.self  // true

// Использование метатипа для создания экземпляров
func create<T: Animal>(_ type: T.Type) -> T where T: NSObject {
    type.init()
}

// С Decodable
func decode<T: Decodable>(_ type: T.Type, from data: Data) throws -> T {
    try JSONDecoder().decode(type, from: data)
}

let user = try decode(User.self, from: jsonData)
```

### Protocol metatypes

```swift
let protocolType: Drawable.Type = Circle.self
// .Protocol — метатип самого протокола
let existentialType: any Drawable.Type = Circle.self

// Проверка
func supports(_ protocol: Any.Type) -> Bool {
    protocol is Codable.Type
}
```

### Dynamic vs Static type

```swift
class Base { }
class Sub: Base { }

let obj: Base = Sub()

// Static type — тип переменной (компилятор видит)
// Base

// Dynamic type — реальный тип экземпляра (runtime)
type(of: obj)  // Sub.Type
```

## Numeric conversion

В Swift **нет неявного** преобразования числовых типов:

```swift
let intVal: Int = 42
let doubleVal: Double = 3.14

// let sum = intVal + doubleVal  // Ошибка! Разные типы

// Явное преобразование
let sum = Double(intVal) + doubleVal  // 45.14

// Int → другие целочисленные
let byte: UInt8 = UInt8(intVal)  // crash если > 255
let byte2 = UInt8(exactly: 300)  // nil (failable)
let byte3 = UInt8(clamping: 300) // 255 (clamped)
let byte4 = UInt8(truncatingIfNeeded: 300)  // 44 (обрезает биты)

// Float ↔ Int
let truncated = Int(3.99)    // 3 (truncation)
let rounded = Int(3.99.rounded())  // 4

// Безопасное преобразование
if let small = Int8(exactly: intVal) {
    print(small)  // 42
}
```

## Сравнение с JS

| Аспект | Swift | JavaScript |
|--------|-------|-----------|
| Type check | `is` (compile-time + runtime) | `typeof`, `instanceof` |
| Safe cast | `as?` → Optional | Нет аналога |
| Force cast | `as!` → crash | Нет (всё "работает") |
| Implicit conversion | Нет | Повсюду (`"5" + 3 = "53"`) |
| Numeric conversion | Только явное | Неявное (`5 + true = 6`) |
| Any | `Any` (explicit) | Всё динамическое |
| Metatype | `Type.self`, `type(of:)` | `constructor`, `typeof` |

```swift
// Swift — безопасное приведение
let value: Any = "42"
if let number = value as? Int {
    print(number)  // не выполнится — String != Int
}
let str = value as! String  // "42"
let num = Int(str)          // Optional(42) — явная конвертация

// JS — неявное приведение
// const value = "42"
// value + 1    // "421" (string concat)
// value - 1    // 41 (numeric coercion)
// value == 42  // true (type coercion)
// value === 42 // false (strict equality)
```
