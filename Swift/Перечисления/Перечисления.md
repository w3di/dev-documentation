## Базовые enum

В Swift `enum` — это полноценный **значимый тип** (value type), не просто набор констант:

```swift
enum Direction {
    case north
    case south
    case east
    case west
}

// Или в одну строку
enum Planet {
    case mercury, venus, earth, mars, jupiter, saturn, uranus, neptune
}

var heading = Direction.north
heading = .south  // тип уже известен — можно опустить имя enum
```

## Raw Values

Каждый case может иметь фиксированное значение определённого типа:

```swift
// Int — автоинкремент
enum StatusCode: Int {
    case ok = 200
    case notFound = 404
    case serverError = 500
}

// String — имя case по умолчанию
enum Fruit: String {
    case apple       // rawValue = "apple"
    case banana      // rawValue = "banana"
    case cherry = "вишня"  // кастомное значение
}

// Использование
let code = StatusCode.ok
print(code.rawValue)  // 200

// Инициализация из raw value (failable)
let status = StatusCode(rawValue: 404)  // Optional(.notFound)
let unknown = StatusCode(rawValue: 999)  // nil
```

## Associated Values

Каждый case может хранить **разные** данные разных типов:

```swift
enum NetworkResult {
    case success(data: Data, statusCode: Int)
    case failure(error: Error, retryAfter: TimeInterval?)
    case loading(progress: Double)
}

let result = NetworkResult.success(data: someData, statusCode: 200)

// Извлечение associated values
switch result {
case .success(let data, let code):
    print("Got \(data.count) bytes, status \(code)")
case .failure(let error, let retry):
    print("Error: \(error), retry after: \(retry ?? 0)")
case .loading(let progress):
    print("Loading: \(Int(progress * 100))%")
}

// if case для одного варианта
if case .success(let data, _) = result {
    print("Success with \(data.count) bytes")
}
```

### Raw Values vs Associated Values

| Raw Values | Associated Values |
|-----------|------------------|
| Все cases одного типа | Каждый case — свой тип |
| Фиксированные значения | Значения при создании экземпляра |
| Автоматическая генерация | Ручное задание |
| Нельзя комбинировать с associated values | Можно |

## Рекурсивные enum

Для рекурсивных структур данных — ключевое слово `indirect`:

```swift
// indirect на отдельных cases
enum ArithExpr {
    case number(Int)
    indirect case addition(ArithExpr, ArithExpr)
    indirect case multiplication(ArithExpr, ArithExpr)
}

// Или indirect на весь enum
indirect enum LinkedList<T> {
    case empty
    case node(value: T, next: LinkedList<T>)
}

// Использование
let expr = ArithExpr.addition(
    .number(5),
    .multiplication(.number(3), .number(4))
)

// (5 + (3 * 4)) = 17
func evaluate(_ expr: ArithExpr) -> Int {
    switch expr {
    case .number(let n):
        return n
    case .addition(let left, let right):
        return evaluate(left) + evaluate(right)
    case .multiplication(let left, let right):
        return evaluate(left) * evaluate(right)
    }
}

let list = LinkedList.node(value: 1,
    next: .node(value: 2,
        next: .node(value: 3,
            next: .empty)))
```

> `indirect` заставляет компилятор хранить данные через **указатель на heap** (box), иначе рекурсивный тип имел бы бесконечный размер.

## Методы и свойства

```swift
enum Suit: String, CaseIterable {
    case hearts = "♥️"
    case diamonds = "♦️"
    case clubs = "♣️"
    case spades = "♠️"

    // Computed property
    var color: String {
        switch self {
        case .hearts, .diamonds: return "red"
        case .clubs, .spades: return "black"
        }
    }

    // Метод
    func beats(_ other: Suit) -> Bool {
        // какая-то логика
        return self == .spades && other != .spades
    }

    // Static метод
    static func random() -> Suit {
        allCases.randomElement()!
    }

    // Mutating метод
    mutating func next() {
        let cases = Suit.allCases
        let index = cases.firstIndex(of: self)!
        let nextIndex = (index + 1) % cases.count
        self = cases[nextIndex]
    }
}

var suit = Suit.hearts
print(suit.color)       // "red"
print(suit.rawValue)    // "♥️"
suit.next()             // .diamonds
```

## CaseIterable

```swift
enum Season: CaseIterable {
    case spring, summer, autumn, winter
}

// Автоматически получаем allCases
for season in Season.allCases {
    print(season)
}

print(Season.allCases.count)  // 4

// ⚠️ Не работает автоматически с associated values
// Нужна ручная реализация:
enum Filter: CaseIterable {
    case all
    case active
    case completed

    // Если бы был case custom(String) — нужно вручную:
    // static var allCases: [Filter] { [.all, .active, .completed] }
}
```

## Pattern matching

```swift
enum Barcode {
    case upc(Int, Int, Int, Int)
    case qrCode(String)
}

let product = Barcode.upc(8, 85909, 51226, 3)

// switch
switch product {
case .upc(let system, let manufacturer, let product, let check):
    print("UPC: \(system), \(manufacturer), \(product), \(check)")
case .qrCode(let code):
    print("QR: \(code)")
}

// Сокращённый let
switch product {
case let .upc(s, m, p, c):  // let перед case
    print("\(s)-\(m)-\(p)-\(c)")
case let .qrCode(code):
    print(code)
}

// Where clause
switch product {
case .upc(let s, _, _, _) where s == 0:
    print("US product")
case .upc:
    print("Other UPC product")
case .qrCode:
    print("QR product")
}

// if case
if case .qrCode(let code) = product {
    print("QR: \(code)")
}

// for case
let items: [Barcode] = [.upc(0, 1, 2, 3), .qrCode("abc"), .upc(1, 2, 3, 4)]
for case .qrCode(let code) in items {
    print(code)  // "abc"
}
```

## Размер в памяти

### Tag + Payload

```swift
// Простой enum без данных — 1 байт (tag byte)
enum Direction { case n, s, e, w }
MemoryLayout<Direction>.size   // 1 (4 варианта = 2 бита, но мин. 1 байт)

// Enum с одним case — 0 байт (нет неоднозначности)
enum Unit { case value }
MemoryLayout<Unit>.size  // 0

// Enum без cases (Never) — 0 байт
enum Never {}
MemoryLayout<Never>.size  // 0
```

### Single-payload enum

Когда только один case имеет associated value:

```swift
enum OptionalInt {
    case none
    case some(Int)
}
MemoryLayout<OptionalInt>.size    // 9 (8 payload + 1 tag)
// Но Optional<Int> тоже 9 — та же оптимизация

// Для ссылочных типов — spare bit optimization
MemoryLayout<AnyObject?>.size     // 8 (nil = нулевой указатель, без tag)
```

### Multi-payload enum

```swift
enum Either {
    case left(Int)      // 8 байт
    case right(String)  // 16 байт
}
MemoryLayout<Either>.size    // 17 (16 max payload + 1 tag)
MemoryLayout<Either>.stride  // 24 (с выравниванием)
```

### @frozen

```swift
@frozen  // Гарантия ABI: не будут добавлены новые cases
enum CompassPoint {
    case north, south, east, west
}
// Компилятор может оптимизировать switch без default
// Используется в стандартной библиотеке (Optional, Result)
```

## Паттерны

### State Machine

```swift
enum ConnectionState {
    case disconnected
    case connecting(attempt: Int)
    case connected(session: String)
    case disconnecting(reason: String)

    mutating func connect() {
        switch self {
        case .disconnected:
            self = .connecting(attempt: 1)
        case .connecting(let attempt):
            self = .connecting(attempt: attempt + 1)
        default:
            break
        }
    }

    mutating func didConnect(session: String) {
        guard case .connecting = self else { return }
        self = .connected(session: session)
    }
}
```

### Result-подобный тип

```swift
enum Validated<T> {
    case valid(T)
    case invalid(errors: [String])

    func map<U>(_ transform: (T) -> U) -> Validated<U> {
        switch self {
        case .valid(let value):
            return .valid(transform(value))
        case .invalid(let errors):
            return .invalid(errors: errors)
        }
    }
}
```

### Namespace

```swift
enum Constants {
    static let apiBaseURL = "https://api.example.com"
    static let timeout: TimeInterval = 30

    enum Colors {
        static let primary = "#007AFF"
        static let secondary = "#5856D6"
    }
}
// Constants нельзя инстанцировать (нет cases)
// let c = Constants()  // Ошибка!
```

## Сравнение с JS

| Аспект | Swift enum | JS/TS |
|--------|-----------|-------|
| Нативная поддержка | Да, first-class type | Нет (объекты/const) |
| Associated values | Да | Нет |
| Pattern matching | `switch` exhaustive | `switch` — нет проверки |
| Raw values | Да (Int, String, etc.) | Объект-маппинг |
| Методы | Да | Нет |
| Value type | Да | — |
| TS enum | — | Числовой/строковый, без associated values |

```swift
// Swift
enum Shape {
    case circle(radius: Double)
    case rectangle(width: Double, height: Double)

    var area: Double {
        switch self {
        case .circle(let r): return .pi * r * r
        case .rectangle(let w, let h): return w * h
        }
    }
}

// В JS/TS для подобного нужен discriminated union:
// type Shape =
//   | { type: 'circle'; radius: number }
//   | { type: 'rectangle'; width: number; height: number }
```
