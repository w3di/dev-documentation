## String как коллекция

`String` в Swift — это коллекция `Character`, где каждый `Character` — один Extended Grapheme Cluster:

```swift
let greeting = "Hello, 🌍!"
print(greeting.count)      // 9 (символов, как видит человек)

for char in greeting {
    print(char)  // H, e, l, l, o, ,,  , 🌍, !
}

// String соответствует Collection, BidirectionalCollection, RangeReplaceableCollection
let first = greeting.first       // Optional("H")
let last = greeting.last         // Optional("!")
let isEmpty = greeting.isEmpty   // false
```

## Unicode Views

Строка предоставляет три различных представления:

```swift
let emoji = "🇷🇺"  // flag: Regional Indicator R + Regional Indicator U

// Character view — 1 символ
emoji.count                    // 1

// Unicode Scalar View — 2 скаляра
emoji.unicodeScalars.count     // 2
for scalar in emoji.unicodeScalars {
    print("U+\(String(scalar.value, radix: 16, uppercase: true))")
    // U+1F1F7, U+1F1FA
}

// UTF-8 View — 8 байт
emoji.utf8.count               // 8

// UTF-16 View — 4 code units
emoji.utf16.count              // 4
```

### Когда что использовать

| View | Когда |
|------|-------|
| `String` (Characters) | Работа с текстом как видит пользователь |
| `unicodeScalars` | Работа с Unicode code points |
| `utf8` | Сериализация, сетевые протоколы, файлы |
| `utf16` | Interop с Objective-C/Foundation |

## Extended Grapheme Clusters

Один `Character` может состоять из множества Unicode-скаляров:

```swift
// Combining marks
let eAcute: Character = "\u{65}\u{301}"  // e + ́ = é
let ePrecomposed: Character = "\u{E9}"    // é (precomposed)
print(eAcute == ePrecomposed)  // true! (canonical equivalence)

// Emoji с модификатором тона кожи
let wave: Character = "\u{1F44B}\u{1F3FD}"  // 👋🏽
wave.unicodeScalars.count  // 2

// Семейный emoji — до 7 скаляров
let family: Character = "👨‍👩‍👧‍👦"
family.unicodeScalars.count  // 7 (4 человека + 3 ZWJ)

// Всё это — ОДИН Character
```

### Каноническая эквивалентность

Swift сравнивает строки по **канонической эквивалентности** (Unicode Normalization Form):

```swift
let a = "\u{00E9}"       // é (precomposed)
let b = "\u{0065}\u{0301}"  // e + combining accent

a == b           // true
a.count == b.count  // true (оба = 1 Character)

// Но UTF-8 представления разные!
Array(a.utf8)  // [195, 169]      — 2 байта
Array(b.utf8)  // [101, 204, 129] — 3 байта
```

## String.Index

В Swift строки **не поддерживают доступ по Int-индексу** — потому что символы имеют разную длину в байтах:

```swift
let text = "Привет"

// text[0]  // Ошибка компиляции!

// Используй String.Index
let start = text.startIndex
let second = text.index(after: start)
let third = text.index(start, offsetBy: 2)

print(text[start])   // "П"
print(text[second])  // "р"
print(text[third])   // "и"

// Диапазон
let range = start..<third
print(text[range])  // "Пр"

// Последний символ
let lastIndex = text.index(before: text.endIndex)
print(text[lastIndex])  // "т"
```

### Почему нет Int-индексации?

`text[5]` подразумевает O(1), но для UTF-8 строк с переменной длиной символов это O(n) — нужно пройти от начала, считая grapheme clusters. Swift делает эту стоимость **явной** через `String.Index`.

```swift
// O(n) — честно показывает стоимость
let index = text.index(text.startIndex, offsetBy: 5)

// Если нужна частая индексация по Int — конвертируй в Array
let chars = Array(text)  // [Character]
print(chars[5])  // O(1), но O(n) памяти
```

## Substring

`Substring` — это view на оригинальную строку (разделяет буфер):

```swift
let greeting = "Hello, World!"
let comma = greeting.firstIndex(of: ",")!
let hello: Substring = greeting[greeting.startIndex..<comma]

print(hello)  // "Hello"
print(type(of: hello))  // Substring

// Substring держит ссылку на ВЕСЬ оригинальный буфер
// Для долговременного хранения — конвертируй в String
let stored = String(hello)  // отдельная копия
```

### Copy-on-Write

```swift
var str1 = "Hello"
var str2 = str1     // Разделяют буфер (COW)

str2 += " World"    // str2 получает свой буфер
// str1 не изменился
```

## String Interpolation

### Базовая интерполяция

```swift
let name = "Alice"
let age = 30
print("Name: \(name), age: \(age)")
// "Name: Alice, age: 30"

// Выражения
print("Next year: \(age + 1)")
print("Uppercase: \(name.uppercased())")
```

### Кастомная интерполяция (ExpressibleByStringInterpolation)

```swift
struct SQLQuery: ExpressibleByStringInterpolation {
    var sql: String
    var bindings: [Any]

    init(stringLiteral value: String) {
        sql = value
        bindings = []
    }

    init(stringInterpolation: StringInterpolation) {
        sql = stringInterpolation.sql
        bindings = stringInterpolation.bindings
    }

    struct StringInterpolation: StringInterpolationProtocol {
        var sql = ""
        var bindings: [Any] = []

        init(literalCapacity: Int, interpolationCount: Int) {
            sql.reserveCapacity(literalCapacity)
            bindings.reserveCapacity(interpolationCount)
        }

        mutating func appendLiteral(_ literal: String) {
            sql += literal
        }

        mutating func appendInterpolation(param value: Any) {
            sql += "?"
            bindings.append(value)
        }
    }
}

let name = "Alice"
let query: SQLQuery = "SELECT * FROM users WHERE name = \(param: name)"
print(query.sql)       // "SELECT * FROM users WHERE name = ?"
print(query.bindings)  // ["Alice"]
```

## Multiline и Raw strings

### Multiline

```swift
let poem = """
    Roses are red,
    Violets are blue,
    Swift is great,
    And so are you.
    """
// Отступ определяется позицией закрывающих """
// Ведущие пробелы до уровня """ удаляются
```

### Raw strings

```swift
let regex = #"(\d+)\s+(\w+)"#
// Нет необходимости экранировать \

let path = #"C:\Users\name\file.txt"#

// Интерполяция в raw strings — #()
let name = "World"
let greeting = #"Hello, \#(name)!"#  // "Hello, World!"

// Множественные #
let code = ##"She said #"hello"#"##
```

## Производительность

### Small String Optimization (SSO)

На 64-bit платформах String использует **16 байт** inline storage. Строки до ~15 UTF-8 байт хранятся **без heap-аллокации**:

```swift
// Маленькая строка — inline (на стеке)
let small = "Hello"  // 5 байт UTF-8 — помещается inline

// Большая строка — heap аллокация
let big = "This is a longer string that exceeds inline storage"
```

### Native vs Bridged

```swift
// Native Swift String — UTF-8 буфер
let swift = "Hello"

// Bridged от NSString — UTF-16 буфер (lazy conversion)
let ns = NSString(string: "Hello") as String
```

При interop с Objective-C могут возникать **bridging costs** — конвертация между UTF-8 и UTF-16.

### Советы по производительности

```swift
// reserveCapacity для конкатенации
var result = ""
result.reserveCapacity(1000)
for i in 0..<100 {
    result += "item \(i), "
}

// Используй Substring вместо создания копий
let csv = "name,age,city"
let fields = csv.split(separator: ",")  // [Substring] — без копирования

// UTF-8 view для byte-level операций
let data = Array(text.utf8)  // [UInt8] — быстрый доступ
```

## Методы

### Поиск и проверка

```swift
let text = "Hello, World!"

text.contains("World")          // true
text.hasPrefix("Hello")         // true
text.hasSuffix("!")             // true
text.isEmpty                    // false

// Поиск индекса
text.firstIndex(of: "W")       // String.Index
text.range(of: "World")        // Range<String.Index>?

// Регистронезависимый поиск
text.range(of: "world", options: .caseInsensitive)
```

### Трансформации

```swift
let text = "  Hello, World!  "

text.uppercased()               // "  HELLO, WORLD!  "
text.lowercased()               // "  hello, world!  "
text.capitalized                // "  Hello, World!  "

text.trimmingCharacters(in: .whitespaces)  // "Hello, World!"

text.replacingOccurrences(of: "World", with: "Swift")

// Split
"a,b,c".split(separator: ",")  // ["a", "b", "c"] — [Substring]

// Join
["a", "b", "c"].joined(separator: ", ")  // "a, b, c"
```

### Модификация

```swift
var str = "Hello"

str.append(" World")           // "Hello World"
str.insert("!", at: str.endIndex)  // "Hello World!"

let range = str.startIndex..<str.index(str.startIndex, offsetBy: 5)
str.replaceSubrange(range, with: "Hi")  // "Hi World!"

str.removeAll { $0 == "!" }   // "Hi World"
```

## Сравнение с JS

| Аспект | Swift String | JavaScript String |
|--------|-------------|------------------|
| Кодировка | UTF-8 (native) | UTF-16 |
| Единица | Character (grapheme cluster) | UTF-16 code unit |
| Индексация | `String.Index` (O(n)) | `number` (O(1), но некорректно для emoji) |
| Мутабельность | `var` — мутабельна | Всегда immutable |
| Equality | Каноническая эквивалентность | Побайтовое сравнение |
| Emoji length | `"🇷🇺".count` → 1 | `"🇷🇺".length` → 4 |
| Substring | `Substring` (shared buffer) | Копия (обычно) |
| Interpolation | `"\(expr)"` | `` `${expr}` `` |
| Value/Ref | Value type (COW) | Примитив (immutable) |

```swift
// Swift — корректный подсчёт
"👨‍👩‍👧‍👦".count  // 1

// JavaScript — некорректный
// "👨‍👩‍👧‍👦".length → 11 (UTF-16 code units)
```
