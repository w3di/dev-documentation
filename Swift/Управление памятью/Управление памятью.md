## ARC

**Automatic Reference Counting** — компилятор вставляет `retain` (+1) и `release` (-1) автоматически:

```swift
class Dog {
    let name: String
    init(name: String) {
        self.name = name
        print("\(name) init")
    }
    deinit { print("\(name) deinit") }
}

var a: Dog? = Dog(name: "Rex")   // RC = 1
var b = a                         // RC = 2
var c = a                         // RC = 3

a = nil  // RC = 2
b = nil  // RC = 1
c = nil  // RC = 0 → deinit → "Rex deinit"
```

### ARC vs GC

| | ARC (Swift) | GC (JavaScript V8) |
|--|------------|-------------------|
| Когда работает | Compile-time (вставка retain/release) | Runtime (tracing) |
| Деструктор | Детерминированный (deinit) | Недетерминированный |
| Паузы | Нет (нет stop-the-world) | Есть (минимизированы) |
| Циклы | Не собирает (нужен weak/unowned) | Собирает автоматически |
| Overhead | Атомарные инкременты/декременты | Tracing + memory pressure |

## Strong references

По умолчанию все ссылки — strong:

```swift
class Node {
    var value: Int
    var next: Node?

    init(value: Int) { self.value = value }
    deinit { print("Node \(value) freed") }
}

var head: Node? = Node(value: 1)
head?.next = Node(value: 2)
head?.next?.next = Node(value: 3)

head = nil
// Node 1 freed → RC next=0 → Node 2 freed → Node 3 freed
// Каскадная деаллокация
```

## Weak references

`weak` — не увеличивает RC, автоматически становится `nil`:

```swift
class Person {
    var name: String
    var apartment: Apartment?
    init(name: String) { self.name = name }
    deinit { print("\(name) deinit") }
}

class Apartment {
    let unit: String
    weak var tenant: Person?  // weak — разрывает цикл
    init(unit: String) { self.unit = unit }
    deinit { print("Apt \(unit) deinit") }
}

var john: Person? = Person(name: "John")
var apt: Apartment? = Apartment(unit: "4A")

john?.apartment = apt
apt?.tenant = john

john = nil  // "John deinit" — weak не удерживает
// apt?.tenant теперь nil (автоматически)
apt = nil   // "Apt 4A deinit"
```

### Правила weak:
- Только `var` (не `let`)
- Всегда `Optional`
- Только для ссылочных типов (`AnyObject`)

## Unowned references

`unowned` — не увеличивает RC, **не** становится nil (crash при доступе к освобождённому):

```swift
class Customer {
    let name: String
    var card: CreditCard?
    init(name: String) { self.name = name }
}

class CreditCard {
    let number: UInt64
    unowned let customer: Customer  // customer всегда живёт дольше

    init(number: UInt64, customer: Customer) {
        self.number = number
        self.customer = customer
    }
}

let john = Customer(name: "John")
john.card = CreditCard(number: 1234_5678, customer: john)
// john.card → CreditCard → john (unowned, не увеличивает RC)
```

### unowned(unsafe)

```swift
// Без проверок — как dangling pointer в C. Максимальная производительность.
unowned(unsafe) var delegate: Delegate
// ⚠️ Доступ к освобождённому объекту = undefined behavior
```

### Когда weak, когда unowned?

| | `weak` | `unowned` |
|--|-------|----------|
| Может быть nil | Да | Нет |
| Тип | `Optional` | Non-optional |
| При деаллокации | Становится nil | Crash (если обратиться) |
| Используй когда | Время жизни неизвестно | Гарантируешь, что объект живёт дольше |
| Overhead | Side table lookup | Нет (но проверка в debug) |

## Retain cycles

Циклическая ссылка — объекты удерживают друг друга, ARC не может их освободить:

```swift
// ⚠️ RETAIN CYCLE
class A {
    var b: B?
    deinit { print("A deinit") }
}

class B {
    var a: A?  // strong → цикл!
    deinit { print("B deinit") }
}

var a: A? = A()
var b: B? = B()
a?.b = b
b?.a = a

a = nil  // A НЕ освободится (B держит strong ref)
b = nil  // B НЕ освободится (A держит strong ref)
// Утечка памяти!
```

### Решение: weak или unowned

```swift
class B {
    weak var a: A?  // weak → цикл разорван
}
```

## Capture lists

Замыкания захватывают ссылки **strong** по умолчанию:

```swift
// ⚠️ Retain cycle через замыкание
class ViewModel {
    var data: [String] = []
    var onUpdate: (() -> Void)?

    func setup() {
        onUpdate = {
            // self захвачен strong → retain cycle
            // self → onUpdate → closure → self
            print(self.data)
        }
    }
}
```

### [weak self]

```swift
class ViewModel {
    var data: [String] = []
    var onUpdate: (() -> Void)?

    func setup() {
        onUpdate = { [weak self] in
            guard let self else { return }
            print(self.data)
        }
    }
}
```

### [unowned self]

```swift
class ViewModel {
    func startTimer() {
        // Timer живёт столько же, сколько ViewModel
        timer = Timer.scheduledTimer(withTimeInterval: 1, repeats: true) { [unowned self] _ in
            self.tick()  // crash если self уже освобождён
        }
    }
}
```

### Capture by value

```swift
var x = 10
let closure = { [x] in  // захватывает КОПИЮ x (значение 10)
    print(x)  // всегда 10
}
x = 20
closure()  // 10 (не 20)

// Без capture list — захват по ссылке
let closure2 = {
    print(x)  // захватывает переменную x
}
x = 30
closure2()  // 30
```

## Value types

Value types (struct, enum) **не используют reference counting** — они копируются:

```swift
struct Point {
    var x, y: Double
}

var a = Point(x: 1, y: 2)
var b = a  // полная копия, нет RC
b.x = 10
// a.x == 1, b.x == 10

// НО: struct со ссылочным свойством — RC для этого свойства
struct Container {
    var label: String        // value type (COW)
    var object: SomeClass    // reference counted!
}
```

## COW Internals

Стандартные коллекции используют reference-counted буфер + COW:

```swift
var a = [1, 2, 3]
// a → Buffer(refcount=1, [1, 2, 3])

var b = a
// a → Buffer(refcount=2, [1, 2, 3]) ← b

b.append(4)
// Мутация! refcount > 1 → копирование буфера
// a → Buffer(refcount=1, [1, 2, 3])
// b → Buffer(refcount=1, [1, 2, 3, 4])
```

### isKnownUniquelyReferenced

```swift
final class Ref<T> {
    var value: T
    init(_ value: T) { self.value = value }
}

struct COW<T> {
    private var ref: Ref<T>

    init(_ value: T) { ref = Ref(value) }

    var value: T {
        get { ref.value }
        set {
            if !isKnownUniquelyReferenced(&ref) {
                ref = Ref(newValue)  // копируем только если shared
            } else {
                ref.value = newValue  // мутируем in-place
            }
        }
    }
}
```

## Object header

Каждый объект класса в Swift имеет header (16 байт на 64-bit):

```
┌──────────────────────────┐
│  Metadata pointer (8B)   │  → type metadata, V-table
├──────────────────────────┤
│  Reference counts (8B)   │  strong RC + unowned RC + flags
├──────────────────────────┤
│  [Side table ptr]        │  (только если есть weak refs)
├──────────────────────────┤
│  Stored properties       │
└──────────────────────────┘
```

### Reference count layout (inline)

```
63                              0
[strong RC (30 bits)][unowned RC (30 bits)][flags (4 bits)]

Flags:
- isDeiniting
- hasSideTable  → если 1, вместо inline RC — указатель на side table
```

### Side tables

Создаются **лениво** при первом weak reference:

```swift
let obj = MyClass()
weak var w = obj  // Создаётся side table

// Side table содержит:
// - Strong RC
// - Unowned RC
// - Weak RC
// - Указатель на объект
```

Когда strong RC = 0:
1. Вызывается `deinit`
2. Если unowned RC > 0 — объект помечен как freed (доступ → crash)
3. Если weak RC > 0 — side table остаётся (weak refs → nil)
4. Когда weak RC = 0 — side table освобождается

## Unsafe pointers

Для работы с памятью напрямую и interop с C:

```swift
// UnsafePointer<T> — read-only typed pointer
// UnsafeMutablePointer<T> — read-write typed pointer
// UnsafeRawPointer — read-only untyped
// UnsafeMutableRawPointer — read-write untyped
// UnsafeBufferPointer<T> — typed buffer (pointer + count)

// Выделение памяти
let ptr = UnsafeMutablePointer<Int>.allocate(capacity: 3)
ptr.initialize(repeating: 0, count: 3)

ptr[0] = 10
ptr[1] = 20
ptr[2] = 30

let buffer = UnsafeBufferPointer(start: ptr, count: 3)
let array = Array(buffer)  // [10, 20, 30]

ptr.deinitialize(count: 3)
ptr.deallocate()

// withUnsafePointer — безопасный scope
var value = 42
withUnsafePointer(to: &value) { ptr in
    print(ptr.pointee)  // 42
}

// MemoryLayout
MemoryLayout<Int>.size       // 8
MemoryLayout<Int>.alignment  // 8
MemoryLayout<Int>.stride     // 8

// withExtendedLifetime — гарантирует, что объект жив
withExtendedLifetime(someObject) {
    let ptr = Unmanaged.passUnretained(someObject).toOpaque()
    callCFunction(ptr)
}
```

## Сравнение с JS

| Аспект | Swift ARC | JavaScript GC |
|--------|----------|--------------|
| Тип | Reference counting | Tracing (Mark-Sweep) |
| Когда | Compile-time вставка | Runtime (периодически) |
| Циклы | Нужен weak/unowned | Собирает автоматически |
| Деструктор | `deinit` (детерминированный) | `FinalizationRegistry` (не гарантирован) |
| Паузы | Нет | Минимальные (incremental) |
| Weak refs | `weak var` (автоматический nil) | `WeakRef` + `deref()` |
| Memory leaks | Retain cycles | Forgotten listeners, closures |
| Manual | Unsafe pointers (opt-in) | Нет |
| Cost per ref | Атомарный incr/decr | Нет (GC трассирует) |

```swift
// Swift — детерминированное освобождение
do {
    let file = FileHandle(path: "/tmp/data")
    // работа с файлом
}  // deinit → файл закрыт СРАЗУ

// JS — недетерминированное
// {
//     const file = openFile("/tmp/data");
//     // работа с файлом
// }
// file может быть собран GC... когда-нибудь
// Поэтому в JS используют try/finally или using
```
